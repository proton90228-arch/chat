/**
 * Лепестки сакуры — точный порт `SakuraBackground.java`.
 *
 * Та же последовательность `java.util.Random(0x56414c4b)`, те же слои глубины, порывы ветра,
 * плотность по площади и «пиксельная» форма из четырёх прямоугольников. Тест `sakura.test.ts`
 * сверяет состояние с эталоном, снятым с оригинального Java-кода (`tools/golden`).
 */
import { JavaRandom } from './javaRandom';

export const SEED = 0x56414c4b; // "VALK"
export const LEN = 7.0;
export const WIDTH = 3.2;
const AREA_PER = 5000;
const DENSITY = 0.55;

/** Число лепестков: `max(10, min(140, round((float)(w*h/5000*0.55))))`. */
export function petalCount(width: number, height: number): number {
  return Math.max(10, Math.min(140, Math.round(Math.fround(((width * height) / AREA_PER) * DENSITY))));
}

export class Petal {
  x: number;
  y: number;
  readonly vy: number;
  readonly vx: number;
  readonly wavePhase: number;
  readonly waveAmp: number;
  rotation: number;
  readonly rotationSpeed: number;
  readonly flickerPhase: number;
  readonly flickerSpeed: number;
  readonly depth: number;
  readonly pale: boolean;

  constructor(
    private readonly random: JavaRandom,
    readonly screenWidth: number,
    readonly screenHeight: number,
  ) {
    const layer = random.nextDouble();
    this.depth = layer < 0.58 ? 0.55 : layer < 0.9 ? 1.0 : 1.45;
    this.x = random.nextDouble() * screenWidth;
    this.y = random.nextDouble() * screenHeight;
    this.vy = (7.0 + random.nextDouble() * 10.0) * this.depth;
    this.vx = (14.0 + random.nextDouble() * 18.0) * this.depth;
    this.wavePhase = random.nextDouble() * Math.PI * 2.0;
    this.waveAmp = 4.0 + random.nextDouble() * 8.0;
    this.rotation = random.nextDouble() * Math.PI * 2.0;
    this.rotationSpeed = (random.nextDouble() - 0.5) * 0.6;
    this.flickerPhase = random.nextDouble() * Math.PI * 2.0;
    this.flickerSpeed = 0.5 + random.nextDouble() * 0.9;
    this.pale = random.nextDouble() < 0.33;
  }

  update(seconds: number, wind: number, elapsed: number): void {
    this.y += this.vy * seconds;
    this.x += this.vx * wind * seconds;
    this.x += Math.sin(this.wavePhase + elapsed * 1.6) * this.waveAmp * seconds * 0.6;
    this.rotation += this.rotationSpeed * seconds;
    if (this.y > this.screenHeight + LEN) {
      this.y = -LEN;
      this.x =
        this.random.nextDouble() < 0.5
          ? this.random.nextDouble() * this.screenWidth * 0.5
          : this.screenWidth * 0.5 + this.random.nextDouble() * this.screenWidth * 0.5;
      if (this.x > this.screenWidth + LEN) this.x = -LEN;
    }
    if (this.x > this.screenWidth + LEN) {
      this.x = -LEN;
      this.y = this.random.nextDouble() * this.screenHeight;
    }
  }

  /** Прозрачность: мерцание × множитель темы × слой. */
  alpha(elapsed: number, alphaMultiplier: number): number {
    const flicker = 0.35 + 0.22 * ((Math.sin(this.flickerPhase + elapsed * this.flickerSpeed) + 1.0) * 0.5);
    const layerAlpha = this.depth < 0.7 ? 0.55 : this.depth > 1.2 ? 0.62 : 1.0;
    return flicker * alphaMultiplier * layerAlpha;
  }
}

/** Прямоугольники лепестка в локальных координатах: `[x, y, w, h]`. */
export const PETAL_RECTS: readonly (readonly [number, number, number, number])[] = (() => {
  const half = LEN * 0.5;
  const edge = Math.ceil(WIDTH * 0.3);
  const tip = Math.ceil(half * 1.4) - half;
  return [
    [-WIDTH * 0.5, 0, WIDTH, half],
    [-WIDTH * 0.5, -half, WIDTH, half],
    [-edge, half, edge * 2, tip],
    [-edge, -Math.ceil(half * 1.4), edge * 2, tip],
  ] as const;
})();

export interface PetalPaint {
  pale: string;
  deep: string;
  /** Множитель прозрачности темы: 0.55 светлая, 0.62 тёмная. */
  alpha: number;
}

export class SakuraSim {
  petals: Petal[] = [];
  reducedMotion = false;
  private readonly random = new JavaRandom(SEED);
  private lastWidth = -1;
  private lastHeight = -1;
  private elapsedSeconds = 0;
  private gustBoost = 0;

  get elapsed(): number {
    return this.elapsedSeconds;
  }

  /** `rebuildIfNeeded`: пересоздаёт лепестки при смене целочисленного размера. */
  resize(width: number, height: number): boolean {
    const w = Math.round(width);
    const h = Math.round(height);
    if (w <= 0 || h <= 0 || (w === this.lastWidth && h === this.lastHeight)) return false;
    this.lastWidth = w;
    this.lastHeight = h;
    const count = petalCount(w, h);
    this.petals = Array.from({ length: count }, () => new Petal(this.random, w, h));
    return true;
  }

  setReducedMotion(reduced: boolean): void {
    this.reducedMotion = reduced;
  }

  /** Разовый порыв ветра (Play — 1.0, наведение — 0.35); затухает за ~1.5 с. */
  gust(strength: number): void {
    if (this.reducedMotion) return;
    this.gustBoost = Math.min(1.6, this.gustBoost + Math.max(0, strength));
  }

  /** Шаг симуляции; `dt` в секундах, ограничивается 0.05 как в Java. */
  step(dt: number): void {
    const seconds = Math.min(0.05, dt);
    this.elapsedSeconds += seconds;
    const motion = this.reducedMotion ? 0.08 : 1.0;
    this.gustBoost *= Math.pow(0.08, seconds);
    if (this.gustBoost < 0.01) this.gustBoost = 0;
    const wind = 0.72 + 0.28 * ((Math.sin(this.elapsedSeconds * 0.42) + 1.0) * 0.5) + this.gustBoost * 2.4;
    for (const petal of this.petals) petal.update(seconds * motion, wind, this.elapsedSeconds);
  }

  /** Отрисовка в canvas, координаты в CSS-пикселях, `scale` — devicePixelRatio. */
  draw(ctx: CanvasRenderingContext2D, paint: PetalPaint, scale: number, shape: Path2D): void {
    ctx.setTransform(1, 0, 0, 1, 0, 0);
    ctx.clearRect(0, 0, ctx.canvas.width, ctx.canvas.height);
    for (const petal of this.petals) {
      ctx.globalAlpha = petal.alpha(this.elapsedSeconds, paint.alpha);
      ctx.fillStyle = petal.pale ? paint.pale : paint.deep;
      const k = petal.depth * scale;
      const cos = Math.cos(petal.rotation) * k;
      const sin = Math.sin(petal.rotation) * k;
      ctx.setTransform(cos, sin, -sin, cos, petal.x * scale, petal.y * scale);
      ctx.fill(shape);
    }
    ctx.globalAlpha = 1;
    ctx.setTransform(1, 0, 0, 1, 0, 0);
  }
}

/** Контур лепестка одним путём: без швов сглаживания между прямоугольниками. */
export function petalPath(): Path2D {
  const path = new Path2D();
  for (const [x, y, w, h] of PETAL_RECTS) path.rect(x, y, w, h);
  return path;
}
