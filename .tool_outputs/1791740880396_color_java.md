package javafx.scene.paint;
public class Color {
    private final double r, g, b, a;
    public Color(double r, double g, double b, double a) { this.r = r; this.g = g; this.b = b; this.a = a; }
    public static Color web(String hex) {
        String h = hex.substring(1);
        return new Color(Integer.parseInt(h.substring(0, 2), 16) / 255.0, Integer.parseInt(h.substring(2, 4), 16) / 255.0,
            Integer.parseInt(h.substring(4, 6), 16) / 255.0, 1.0);
    }
    public double getRed() { return r; }
    public double getGreen() { return g; }
    public double getBlue() { return b; }
    public double getOpacity() { return a; }
}
