export type TaskEvent =
  | { kind: 'status'; message: string }
  | { kind: 'stage'; message: string }
  | { kind: 'progress'; fraction: number | null };

const LAUNCH_STATE_IDS: readonly string[] = [
  'CHECKING', 'INSTALL_REQUIRED', 'INSTALLING', 'READY', 'UPDATE_AVAILABLE',
  'UPDATING', 'STARTING', 'RUNNING', 'UNAVAILABLE', 'ERROR',
];

type Shape = Record<string, 'string' | 'number' | 'boolean' | 'string?' | 'number?' | 'object?' | 'array' | 'array?'>;

function matches(value: unknown, shape: Shape): value is Record<string, unknown> {
  if (typeof value !== 'object' || value === null) return false;
  const record = value as Record<string, unknown>;
  return Object.entries(shape).every(([key, kind]) => {
    const field = record[key];
    const optional = kind.endsWith('?');
    if (optional && field === null) return true;
    const base = kind.replace('?', '');
    if (base === 'array') return Array.isArray(field);
    if (base === 'object') return typeof field === 'object' && field !== null;
    return typeof field === base;
  });
}

export function isTaskEvent(value: unknown): value is TaskEvent {
  if (typeof value !== 'object' || value === null) return false;
  const event = value as Record<string, unknown>;
  if (event.kind === 'status' || event.kind === 'stage') return typeof event.message === 'string';
  if (event.kind === 'progress') return event.fraction === null || typeof event.fraction === 'number';
  return false;
}

export function isSettings(value: unknown): value is Settings {
  return (
    matches(value, {
      nickname: 'string', manifestUrl: 'string', memoryMb: 'number', automaticMemory: 'boolean',
      selectedVersion: 'string', selectedValkyrie: 'boolean', javaMode: 'string', customJava: 'string',
      theme: 'string', reducedMotion: 'boolean', uiSounds: 'boolean', changelogOpen: 'boolean', palette: 'string',
    }) &&
    ['automatic', 'custom'].includes(value.javaMode as string) &&
    ['light', 'dark'].includes(value.theme as string)
  );
}

export function isGameVersion(value: unknown): value is GameVersion {
  return matches(value, { id: 'string', type: 'string', metadataUrl: 'string?', metadataSha1: 'string', valkyrie: 'boolean' });
}

export function isClientManifest(value: unknown): value is ClientManifest {
  return matches(value, {
    version: 'string', minecraftVersion: 'string', forgeProfile: 'string', jarUrl: 'string', sha256: 'string', changelog: 'array',
  });
}

export function isStatusReport(value: unknown): value is StatusReport {
  return (
    matches(value, {
      state: 'string', detail: 'string', manifest: 'object?', manifestInstalled: 'boolean',
      changelog: 'array?', warning: 'string?', installedLabel: 'string',
    }) &&
    LAUNCH_STATE_IDS.includes(value.state as string) &&
    (value.manifest === null || isClientManifest(value.manifest))
  );
}
