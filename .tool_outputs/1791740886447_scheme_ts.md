const EXT_LIGHT = { '': 40, 'on-': 100, '-container': 90, 'on--container': 30 } as const;
const EXT_DARK = { '': 80, 'on-': 20, '-container': 30, 'on--container': 90 } as const;

export type Scheme = Record<string, string>;

export interface SchemeSet {
  seed: string;
  light: Scheme;
  dark: Scheme;
}

function extKey(pattern: string, name: string): string {
  if (pattern.includes('--')) return pattern.replace('--', `-${name}-`);
  return pattern.endsWith('-') ? pattern + name : name + pattern;
}

const cache = new Map<string, SchemeSet>();

/** Полная схема (светлая и тёмная) из seed-цвета `#RRGGBB`. */
export function buildScheme(seedHex: string): SchemeSet {
  const seed = seedHex.toUpperCase();
  const cached = cache.get(seed);
  if (cached) return cached;
  const { hue, chroma } = hctFromHex(seed);
  const palettes: Record<PaletteKey, TonalPalette> = {
    P: new TonalPalette(hue, Math.max(chroma, 36.0)),
    S: new TonalPalette(hue, 16.0),
    T: new TonalPalette(hue + 60.0, 24.0),
    N: new TonalPalette(hue, 4.0),
    NV: new TonalPalette(hue, 8.0),
    DN: new TonalPalette(mod360(hue - 28.0), 7.0),
    E: (() => {
      const error = harmonize('#B3261E', seed);
      return new TonalPalette(error.hue, Math.max(error.chroma, 60.0));
    })(),
  };
  const ext = Object.entries(EXT).map(([name, hex]) => {
    const color = NOT_HARMONIZED.has(name) ? hctFromHex(hex) : harmonize(hex, seed);
    return [name, new TonalPalette(color.hue, Math.max(color.chroma, 36.0))] as const;
  });
  const make = (roles: RoleMap, extTones: Record<string, number>): Scheme => {
    const scheme: Scheme = {};
    for (const [role, [palette, tone]] of Object.entries(roles)) scheme[role] = palettes[palette].tone(tone);
    for (const [name, palette] of ext) {
      for (const [pattern, tone] of Object.entries(extTones)) scheme[extKey(pattern, name)] = palette.tone(tone);
    }
    return scheme;
  };
  const result = { seed, light: make(LIGHT, EXT_LIGHT), dark: make(DARK, EXT_DARK) };
  cache.set(seed, result);
  return result;
}
