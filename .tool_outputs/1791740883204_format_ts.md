import { t } from '../i18n/strings';
import type { Settings } from '../ipc/types';

export const NICKNAME_PATTERN = /^[A-Za-z0-9_]{3,16}$/;
export const isValidNickname = (value: string) => NICKNAME_PATTERN.test(value.trim());

/** Ползунок памяти: 2048–8192 МБ, шаг 1024 (как в Java). */
export const MEMORY_MIN = 2048;
export const MEMORY_MAX = 8192;
export const MEMORY_STEP = 1024;

export function snapMemory(value: number): number {
  const snapped = Math.round(value / MEMORY_STEP) * MEMORY_STEP;
  return Math.min(MEMORY_MAX, Math.max(MEMORY_MIN, snapped));
}

export function memoryText(settings: Settings, effectiveMemoryMb: number): string {
  return settings.automaticMemory ? `Auto · ${effectiveMemoryMb} MB` : `${settings.memoryMb} MB`;
}

export function javaText(settings: Settings): string {
  if (settings.javaMode !== 'custom' || !settings.customJava) return t.javaAutomatic;
  const name = settings.customJava.split(/[\\/]/).pop() ?? settings.customJava;
  return `Custom · ${name}`;
}

export function errorText(error: unknown): string {
  if (typeof error === 'string' && error.trim()) return error;
  if (error instanceof Error && error.message) return error.message;
  return t.unknownError;
}
