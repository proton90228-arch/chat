import { describe, expect, it } from 'vitest';
import golden from './fixtures/sakura-golden.json';
import { JavaRandom } from '../src/sakura/javaRandom';
import { PETAL_RECTS, petalCount, SakuraSim } from '../src/sakura/sim';

const LIGHT = { pale: 0.94902, deep: 0.909804, mul: 0.55 };
const DARK = { pale: 0.968627, deep: 0.913725, mul: 0.62 };

describe('java.util.Random', () => {
  it('совпадает с эталонной последовательностью Java', () => {
    const random = new JavaRandom(42);
    // new Random(42).nextInt() ×3 и nextDouble() — значения из JDK 21
    expect([random.nextInt(), random.nextInt(), random.nextInt()]).toEqual([-1170105035, 234785527, -1360544799]);
    expect(random.nextDouble()).toBe(0.047939305137387644);
  });
});

describe('SakuraBackground parity', () => {
  it('число лепестков как в Java', () => {
    expect(petalCount(1040, 650)).toBe(74);
    expect(petalCount(820, 560)).toBe(51);
    expect(petalCount(100, 100)).toBe(10);
    expect(petalCount(4000, 3000)).toBe(140);
  });

  it('геометрия лепестка', () => {
    expect(PETAL_RECTS.map((r) => [...r])).toEqual(golden.rects);
  });

  it('траектории, порывы, смена размера, reduced motion и тёмная тема совпадают с оригиналом', () => {
    const sim = new SakuraSim();
    // Java: слушатели ширины и высоты срабатывают по очереди.
    sim.resize(1040, 0);
    sim.resize(1040, 650);
    const snapshots = new Map(golden.frames.map((f) => [f.frame, f.petals]));
    let colors = LIGHT;
    let checked = 0;
    for (let frame = 1; frame <= 260; frame++) {
      if (frame === 30) sim.gust(1.0);
      if (frame === 90) {
        sim.gust(0.35);
        sim.gust(0.35);
      }
      if (frame === 150) {
        sim.resize(820, 650);
        sim.resize(820, 560);
      }
      if (frame === 200) {
        sim.reducedMotion = true;
        sim.gust(1.0);
      }
      if (frame === 230) {
        sim.reducedMotion = false;
        colors = DARK;
      }
      const deltaNs = golden.pattern[frame % golden.pattern.length]!;
      sim.step(Math.min(0.05, deltaNs / 1_000_000_000));
      const expected = snapshots.get(frame);
      if (!expected) continue;
      expect(sim.petals.length).toBe(expected.length);
      expected.forEach(([x, y, rotation, depth, alpha, red], index) => {
        const petal = sim.petals[index]!;
        expect(petal.x).toBeCloseTo(x!, 6);
        expect(petal.y).toBeCloseTo(y!, 6);
        expect(petal.rotation).toBeCloseTo(rotation!, 6);
        expect(petal.depth).toBe(depth!);
        expect(petal.alpha(sim.elapsed, colors.mul)).toBeCloseTo(alpha!, 9);
        expect(petal.pale ? colors.pale : colors.deep).toBeCloseTo(red!, 5);
      });
      checked++;
    }
    expect(checked).toBe(golden.frames.length);
  });
});
