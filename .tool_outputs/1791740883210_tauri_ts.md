import { Channel, invoke } from '@tauri-apps/api/core';
import { listen } from '@tauri-apps/api/event';
import { getCurrentWindow } from '@tauri-apps/api/window';
import { writeText } from '@tauri-apps/plugin-clipboard-manager';
import { open } from '@tauri-apps/plugin-dialog';
import { check } from '@tauri-apps/plugin-updater';
import type { Backend } from './api';
import type { GameExit, TaskEvent } from './types';

function channel(handler: (event: TaskEvent) => void): Channel<TaskEvent> {
  const result = new Channel<TaskEvent>();
  result.onmessage = handler;
  return result;
}

export function createTauriBackend(): Backend {
  const current = getCurrentWindow();
  return {
    bootstrap: () => invoke('bootstrap'),
    loadVersions: () => invoke('load_versions'),
    selectVersion: (version) => invoke('select_version', { version }),
    checkStatus: () => invoke('check_status'),
    updateSettings: (patch) => invoke('update_settings', { patch }),
    setJava: (mode, path) => invoke('set_java', { mode, path: path ?? null }),
    installAndLaunch: (nickname, onEvent) => invoke('install_and_launch', { nickname, onEvent: channel(onEvent) }),
    repair: (onEvent) => invoke('repair', { onEvent: channel(onEvent) }),
    openFolder: (kind) => invoke('open_folder', { kind }),
    onGameExit: (handler) => listen<GameExit>('game-exited', (event) => handler(event.payload)),
    async pickJava() {
      const picked: unknown = await open({
        multiple: false,
        directory: false,
        title: 'Выберите javaw.exe или java.exe',
        filters: [{ name: 'Java', extensions: ['exe'] }],
      });
      return typeof picked === 'string' ? picked : null;
    },
    systemAccent: () => invoke('system_accent'),
    copyText: (text) => writeText(text),
    log(level, message) {
      void invoke('log_ui', { level, message }).catch(() => undefined);
    },
    async checkLauncherUpdate() {
      const update = await check();
      if (!update) return null;
      return {
        version: update.version,
        notes: update.body ?? '',
        async install(onProgress) {
          let total = 0;
          let received = 0;
          await update.downloadAndInstall((event) => {
            if (event.event === 'Started') {
              total = event.data.contentLength ?? 0;
              onProgress(total ? 0 : null);
            } else if (event.event === 'Progress') {
              received += event.data.chunkLength;
              onProgress(total ? received / total : null);
            } else {
              onProgress(1);
            }
          });
        },
      };
    },
    window: {
      startDragging: () => current.startDragging(),
      minimize: () => current.minimize(),
      close: () => current.close(),
      async show() {
        await current.show();
        await current.setFocus();
      },
    },
  };
}
