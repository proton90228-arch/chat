/**
 * Прогресс кнопки запуска как в Java: в пределах одного запуска значение только растёт;
 * в неопределённых фазах (Forge installer, проверка) полоса медленно «доползает» к 99,5 %.
 */
export class ProgressModel {
  private target = 0;
  private shown = 0;
  private creeping = false;

  /** `null` — неопределённо (в Java `-1`). */
  set(fraction: number | null): void {
    if (fraction === null || fraction < 0) {
      this.creeping = true;
      return;
    }
    this.creeping = false;
    this.target = Math.max(this.target, Math.min(1, fraction));
  }

  reset(): void {
    this.target = 0;
    this.shown = 0;
    this.creeping = false;
  }

  get value(): number {
    return this.shown;
  }

  get indeterminate(): boolean {
    return this.creeping;
  }

  /** Шаг анимации на `dt` секунд; `immediate` — без сглаживания (reduced motion). Возвращает 0…1. */
  tick(dt: number, immediate = false): number {
    if (this.creeping) {
      this.target = Math.min(0.995, this.target + Math.max(0.0004, (0.995 - this.target) * 0.012) * dt);
    }
    const follow = immediate ? 1 : 1 - Math.exp(-dt / 0.18);
    this.shown += (this.target - this.shown) * follow;
    if (Math.abs(this.target - this.shown) < 0.0005) this.shown = this.target;
    return this.shown;
  }
}
