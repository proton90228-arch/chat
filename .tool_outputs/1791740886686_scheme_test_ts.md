import { describe, expect, it } from 'vitest';
import { contrast, hctFromHex, hctToHex } from '../src/theme/hct';
import { buildScheme, PRESETS } from '../src/theme/scheme';
import reference from './fixtures/schemes.json';

type Reference = Record<string, { seed: string; light: Record<string, string>; dark: Record<string, string> }>;
const schemes = reference as unknown as Reference;

describe('HCT', () => {
  it('round-trips sRGB colors', () => {
    for (const hex of ['#4285F4', '#DF91AA', '#6750A4', '#B3261E']) {
      const { hue, chroma, tone } = hctFromHex(hex);
      expect(hctToHex(hue, chroma, tone)).toBe(hex);
    }
    expect(hctToHex(10, 50, 0)).toBe('#000000');
    expect(hctToHex(10, 50, 100)).toBe('#FFFFFF');
  });

  it('matches the Sakura seed from the design tokens', () => {
    const { hue, chroma, tone } = hctFromHex('#DF91AA');
    expect(hue).toBeCloseTo(359.38, 1);
    expect(chroma).toBeCloseTo(35.81, 1);
    expect(tone).toBeCloseTo(68.64, 1);
  });
});

describe('Sakura You schemes', () => {
  for (const [id, preset] of Object.entries(PRESETS)) {
    it(`reproduces design tokens for ${id}`, () => {
      const built = buildScheme(preset.seed);
      const expected = schemes[id]!;
      for (const mode of ['light', 'dark'] as const) {
        expect(Object.keys(built[mode]).sort()).toEqual(Object.keys(expected[mode]).sort());
        for (const [role, hex] of Object.entries(expected[mode])) {
          expect(`${mode}.${role}=${built[mode][role]}`).toBe(`${mode}.${role}=${hex}`);
        }
      }
    });
  }

  it('keeps text readable (WCAG AA)', () => {
    for (const preset of Object.values(PRESETS)) {
      const { light, dark } = buildScheme(preset.seed);
      for (const s of [light, dark]) {
        expect(contrast(s['on-surface']!, s.surface!)).toBeGreaterThan(7);
        expect(contrast(s['on-primary']!, s.primary!)).toBeGreaterThan(4.5);
        expect(contrast(s['on-primary-container']!, s['primary-container']!)).toBeGreaterThan(4.5);
        expect(contrast(s['on-surface-variant']!, s['surface-container']!)).toBeGreaterThan(4.5);
      }
    }
  });
});
