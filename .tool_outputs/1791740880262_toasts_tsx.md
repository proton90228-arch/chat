import { useController, useUi } from '../app/hooks';
import { Icon } from './Icon';

/** Снекбары M3: до трёх, новые сверху, клик закрывает. */
export function Toasts() {
  const controller = useController();
  const toasts = useUi((state) => state.toasts);
  return (
    <div className="toasts" role="status" aria-live="polite">
      {toasts.map((toast) => (
        <button
          key={toast.id}
          type="button"
          className="toast"
          data-error={toast.error}
          data-leaving={toast.leaving}
          onClick={() => controller.dismissToast(toast.id)}
        >
          <Icon name={toast.error ? 'alert' : 'sakura'} size={18} />
          <span>{toast.message}</span>
        </button>
      ))}
    </div>
  );
}
