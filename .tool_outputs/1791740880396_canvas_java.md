package javafx.scene.canvas;
import stub.Prop;
public class Canvas {
    private final Prop width = new Prop();
    private final Prop height = new Prop();
    private final GraphicsContext gc = new GraphicsContext();
    public Prop widthProperty() { return width; }
    public Prop heightProperty() { return height; }
    public double getWidth() { return width.get(); }
    public double getHeight() { return height.get(); }
    public void setWidth(double v) { width.set(v); }
    public void setHeight(double v) { height.set(v); }
    public GraphicsContext getGraphicsContext2D() { return gc; }
    public void setMouseTransparent(boolean value) {}
}
