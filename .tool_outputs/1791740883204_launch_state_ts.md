import { t } from '../i18n/strings';
import type { LaunchStateId } from '../ipc/types';

export type ActionIcon = 'spinner' | 'download' | 'play' | 'update' | 'stop' | 'alert';

export interface LaunchStateInfo {
  /** Текст на кнопке. */
  button: string;
  /** Текст в строке статуса. */
  status: string;
  actionable: boolean;
  busy: boolean;
  icon: ActionIcon;
}

/** `LaunchState` из Java 1:1. */
export const LAUNCH_STATES: Record<LaunchStateId, LaunchStateInfo> = {
  CHECKING: { button: 'Проверяем', status: 'Проверка', actionable: false, busy: true, icon: 'spinner' },
  INSTALL_REQUIRED: { button: 'Установить и запустить', status: 'Требуется установка', actionable: true, busy: false, icon: 'download' },
  INSTALLING: { button: 'Устанавливаем', status: 'Подготовка игры', actionable: false, busy: true, icon: 'spinner' },
  READY: { button: 'Запустить', status: 'Готово', actionable: true, busy: false, icon: 'play' },
  UPDATE_AVAILABLE: { button: 'Обновить и запустить', status: 'Доступно обновление', actionable: true, busy: false, icon: 'update' },
  UPDATING: { button: 'Обновляем', status: 'Обновление', actionable: false, busy: true, icon: 'spinner' },
  STARTING: { button: 'Запускаем', status: 'Запуск', actionable: false, busy: true, icon: 'spinner' },
  RUNNING: { button: 'Запущено', status: 'Игра запущена', actionable: false, busy: false, icon: 'stop' },
  UNAVAILABLE: { button: 'Не установлен', status: 'Клиент недоступен', actionable: false, busy: false, icon: 'alert' },
  ERROR: { button: 'Повторить', status: 'Ошибка', actionable: true, busy: false, icon: 'alert' },
};

/** «Характер» кнопки: установка — спокойный тон, обновление — акцент, запуск — бренд. */
export type PlayTone = 'brand' | 'install' | 'update' | 'error' | 'muted';

export function playTone(state: LaunchStateId): PlayTone {
  switch (state) {
    case 'INSTALL_REQUIRED':
      return 'install';
    case 'UPDATE_AVAILABLE':
    case 'UPDATING':
      return 'update';
    case 'ERROR':
    case 'UNAVAILABLE':
      return 'error';
    case 'RUNNING':
      return 'muted';
    default:
      return 'brand';
  }
}

/** Строка статуса: `Готово · Valkyrie 0.9.0-beta`. */
export function statusLine(state: LaunchStateId, detail: string | null | undefined): string {
  const text = LAUNCH_STATES[state].status;
  return detail && detail.trim() ? `${text} · ${detail}` : text;
}

/** Цвет точки статуса. */
export function statusDot(state: LaunchStateId): 'error' | 'accent' | 'normal' {
  if (state === 'ERROR') return 'error';
  if (state === 'UPDATE_AVAILABLE' || state === 'UPDATING') return 'accent';
  return 'normal';
}

export type BadgeKind = 'ok' | 'update' | 'none' | 'error';

/** Индикатор «что установлено» на карточке версии (`updateInstallBadge`). */
export function installBadge(
  state: LaunchStateId,
  valkyrie: boolean,
  installedLabel: string,
  manifestVersion: string | null,
): { kind: BadgeKind; text: string } {
  switch (state) {
    case 'READY':
    case 'RUNNING':
      return { kind: 'ok', text: valkyrie ? t.installed + installedLabel : t.installedVanilla };
    case 'UPDATE_AVAILABLE':
      return { kind: 'update', text: t.updateAvailable + (manifestVersion ? ` · ${manifestVersion}` : '') };
    case 'INSTALL_REQUIRED':
      return { kind: 'none', text: t.notInstalled };
    case 'ERROR':
      return { kind: 'error', text: t.error };
    default:
      return { kind: 'none', text: '' };
  }
}
