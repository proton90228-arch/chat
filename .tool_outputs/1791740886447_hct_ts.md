/**
 * HCT (Hue · Chroma · Tone) из Material Color Utilities: CAM16 + CIE L*.
 * Стандартные условия просмотра MCU (sRGB, D65, L* фона 50, surround 2).
 * Порт design/hct.py с тем же порядком операций: палитры совпадают с дизайн-токенами.
 */

type Vec3 = [number, number, number];
type Mat3 = readonly (readonly [number, number, number])[];

const WHITE: Vec3 = [95.047, 100.0, 108.883];
const SRGB_TO_XYZ: Mat3 = [
  [0.41233895, 0.35762064, 0.18051042],
  [0.2126, 0.7152, 0.0722],
  [0.01932141, 0.11916382, 0.95034478],
];
const XYZ_TO_SRGB: Mat3 = [
  [3.2413774792388685, -1.5376652402851851, -0.49885366846268053],
  [-0.9691452513005321, 1.8758853451067872, 0.04156585616912061],
  [0.05562093689691305, -0.20395524564742123, 1.0571799111220335],
];
const XYZ_TO_CAM16RGB: Mat3 = [
  [0.401288, 0.650173, -0.051461],
  [-0.250268, 1.204414, 0.045854],
  [-0.002079, 0.048952, 0.953127],
];

const DEG_TO_RAD = Math.PI / 180;
const RAD_TO_DEG = 180 / Math.PI;

function mul(m: Mat3, v: Vec3): Vec3 {
  return [0, 1, 2].map((i) => m[i]![0] * v[0] + m[i]![1] * v[1] + m[i]![2] * v[2]) as Vec3;
}

/** Остаток как в Python: всегда в [0, 360). */
export function mod360(value: number): number {
  const r = value % 360;
  return r < 0 ? r + 360 : r;
}

const sign = (v: number) => (v < 0 ? -1 : 1);

/** Банковское округление, как `round()` в Python. */
function roundHalfEven(x: number): number {
  const r = Math.round(x);
  return Math.abs(x % 1) === 0.5 && r % 2 !== 0 ? r - 1 : r;
}

function labF(t: number): number {
  const e = 216 / 24389;
  const k = 24389 / 27;
  return t > e ? t ** (1 / 3) : (k * t + 16) / 116;
}

function labInvF(ft: number): number {
  const e = 216 / 24389;
  const k = 24389 / 27;
  const ft3 = ft ** 3;
  return ft3 > e ? ft3 : (116 * ft - 16) / k;
}

export const yFromLstar = (l: number) => 100.0 * labInvF((l + 16) / 116);
export const lstarFromY = (y: number) => 116 * labF(y / 100) - 16;

function linearized(c8: number): number {
  const n = c8 / 255.0;
  return (n <= 0.040449936 ? n / 12.92 : ((n + 0.055) / 1.055) ** 2.4) * 100;
}

function delinearized(c: number): number {
  const n = c / 100.0;
  const v = n <= 0.0031308 ? n * 12.92 : 1.055 * n ** (1 / 2.4) - 0.055;
  return Math.max(0, Math.min(255, roundHalfEven(v * 255)));
}

const V = (() => {
  const adapting = ((200.0 / Math.PI) * yFromLstar(50.0)) / 100.0;
  const [rW, gW, bW] = mul(XYZ_TO_CAM16RGB, WHITE);
  const f = 0.8 + 2.0 / 10.0;
  const c = f >= 0.9 ? 0.59 + (0.69 - 0.59) * ((f - 0.9) * 10) : 0.525 + (0.59 - 0.525) * ((f - 0.8) * 10);
  let d = f * (1 - (1 / 3.6) * Math.exp((-adapting - 42) / 92));
  d = Math.max(0, Math.min(1, d));
  const rgbD: Vec3 = [d * (100 / rW) + 1 - d, d * (100 / gW) + 1 - d, d * (100 / bW) + 1 - d];
  const k = 1 / (5 * adapting + 1);
  const k4 = k ** 4;
  const k4F = 1 - k4;
  const fl = k4 * adapting + 0.1 * k4F * k4F * (5 * adapting) ** (1 / 3);
  const n = yFromLstar(50.0) / WHITE[1];
  const z = 1.48 + Math.sqrt(n);
  const nbb = 0.725 / n ** 0.2;
  const fac = [rW, gW, bW].map((w, i) => ((fl * rgbD[i]! * w) / 100) ** 0.42);
  const rgbA = fac.map((x) => (400 * x) / (x + 27.13));
  const aw = (2 * rgbA[0]! + rgbA[1]! + 0.05 * rgbA[2]!) * nbb;
  return { nc: f, c, rgbD, fl, n, z, nbb, ncb: nbb, aw };
})();

function cam16FromXyz(x: number, y: number, z: number): Vec3 {
  const [rC, gC, bC] = mul(XYZ_TO_CAM16RGB, [x, y, z]);
  const rD = V.rgbD[0] * rC;
  const gD = V.rgbD[1] * gC;
  const bD = V.rgbD[2] * bC;
  const adapt = (v: number) => {
    const af = ((V.fl * Math.abs(v)) / 100) ** 0.42;
    return (sign(v) * 400 * af) / (af + 27.13);
  };
  const rA = adapt(rD);
  const gA = adapt(gD);
  const bA = adapt(bD);
  const a = (11 * rA - 12 * gA + bA) / 11;
  const b = (rA + gA - 2 * bA) / 9;
  const u = (20 * rA + 20 * gA + 21 * bA) / 20;
  const p2 = (40 * rA + 20 * gA + bA) / 20;
  const hue = mod360(RAD_TO_DEG * Math.atan2(b, a));
  const ac = p2 * V.nbb;
  const j = 100 * (ac / V.aw) ** (V.c * V.z);
  const hp = hue < 20.14 ? hue + 360 : hue;
  const eHue = 0.25 * (Math.cos(DEG_TO_RAD * hp + 2) + 3.8);
  const p1 = (50000 / 13) * eHue * V.nc * V.ncb;
  const t = (p1 * Math.hypot(a, b)) / (u + 0.305);
  const alpha = t ** 0.9 * (1.64 - 0.29 ** V.n) ** 0.73;
  return [hue, alpha * Math.sqrt(j / 100), j];
}

function xyzFromJch(j: number, chroma: number, hue: number): Vec3 {
  const alpha = chroma === 0 || j === 0 ? 0 : chroma / Math.sqrt(j / 100);
  const t = (alpha / (1.64 - 0.29 ** V.n) ** 0.73) ** (1 / 0.9);
  const h = DEG_TO_RAD * hue;
  const eHue = 0.25 * (Math.cos(h + 2) + 3.8);
  const ac = V.aw * (j / 100) ** (1 / V.c / V.z);
  const p1 = eHue * (50000 / 13) * V.nc * V.ncb;
  const p2 = ac / V.nbb;
  const hs = Math.sin(h);
  const hc = Math.cos(h);
  const gamma = (23 * (p2 + 0.305) * t) / (23 * p1 + 11 * t * hc + 108 * t * hs);
  const a = gamma * hc;
  const b = gamma * hs;
  const rA = (460 * p2 + 451 * a + 288 * b) / 1403;
  const gA = (460 * p2 - 891 * a - 261 * b) / 1403;
  const bA = (460 * p2 - 220 * a - 6300 * b) / 1403;
  const un = (v: number) => {
    const base = Math.max(0, (27.13 * Math.abs(v)) / (400 - Math.abs(v)));
    return sign(v) * (100 / V.fl) * base ** (1 / 0.42);
  };
  const rF = un(rA) / V.rgbD[0];
  const gF = un(gA) / V.rgbD[1];
  const bF = un(bA) / V.rgbD[2];
  return [
    1.86206786 * rF - 1.01125463 * gF + 0.14918677 * bF,
    0.38752654 * rF + 0.62144744 * gF - 0.00897398 * bF,
    -0.0158415 * rF - 0.03412294 * gF + 1.04996444 * bF,
  ];
}

export function hexToRgb(hex: string): Vec3 {
  const h = hex.replace('#', '');
  return [0, 2, 4].map((i) => parseInt(h.slice(i, i + 2), 16)) as Vec3;
}

export function rgbToHex([r, g, b]: Vec3): string {
  return '#' + [r, g, b].map((c) => c.toString(16).padStart(2, '0').toUpperCase()).join('');
}

export interface Hct {
  hue: number;
  chroma: number;
  tone: number;
}

export function hctFromHex(hex: string): Hct {
  const lin = hexToRgb(hex).map(linearized) as Vec3;
  const [x, y, z] = mul(SRGB_TO_XYZ, lin);
  const [hue, chroma] = cam16FromXyz(x, y, z);
  return { hue, chroma, tone: lstarFromY(y) };
}

function solveJForY(hue: number, chroma: number, yTarget: number): number {
  let lo = 0.0;
  let hi = 100.0;
  for (let i = 0; i < 60; i++) {
    const mid = (lo + hi) / 2;
    const y = xyzFromJch(mid, chroma, hue)[1];
    if (y < yTarget) lo = mid;
    else hi = mid;
  }
  return (lo + hi) / 2;
}

function linearRgb(hue: number, chroma: number, tone: number): Vec3 {
  const j = solveJForY(hue, chroma, yFromLstar(tone));
  return mul(XYZ_TO_SRGB, xyzFromJch(j, chroma, hue));
}

const inGamut = (lin: Vec3, eps = 0.01) => lin.every((c) => -eps <= c && c <= 100 + eps);

/** Цвет с заданными hue/tone и максимально достижимой chroma ≤ запрошенной. */
export function hctToHex(hue: number, chroma: number, tone: number): string {
  if (tone <= 0.0001) return '#000000';
  if (tone >= 99.9999) return '#FFFFFF';
  let lin = linearRgb(hue, chroma, tone);
  if (!inGamut(lin)) {
    let lo = 0.0;
    let hi = chroma;
    for (let i = 0; i < 40; i++) {
      const mid = (lo + hi) / 2;
      if (inGamut(linearRgb(hue, mid, tone))) lo = mid;
      else hi = mid;
    }
    lin = linearRgb(hue, lo, tone);
  }
  return rgbToHex(lin.map((c) => delinearized(Math.max(0, Math.min(100, c)))) as Vec3);
}

/** Контраст WCAG 2.x. */
export function contrast(a: string, b: string): number {
  const lum = (hex: string) => {
    const [r, g, bl] = hexToRgb(hex).map((c) => linearized(c) / 100) as Vec3;
    return 0.2126 * r + 0.7152 * g + 0.0722 * bl;
  };
  const [hi, lo] = [lum(a), lum(b)].sort((x, y) => y - x) as [number, number];
  return (hi + 0.05) / (lo + 0.05);
}

export class TonalPalette {
  readonly hue: number;
  private readonly cache = new Map<number, string>();

  constructor(hue: number, readonly chroma: number) {
    this.hue = mod360(hue);
  }

  tone(t: number): string {
    let hex = this.cache.get(t);
    if (!hex) {
      hex = hctToHex(this.hue, this.chroma, t);
      this.cache.set(t, hex);
    }
    return hex;
  }
}

/** M3 Blend.harmonize: поворот hue к источнику на min(15°, разница/2). */
export function harmonize(designHex: string, sourceHex: string): { hex: string; hue: number; chroma: number } {
  const d = hctFromHex(designHex);
  const s = hctFromHex(sourceHex);
  const diff = 180 - Math.abs(Math.abs(d.hue - s.hue) - 180);
  const rotation = Math.min(diff * 0.5, 15.0);
  const direction = mod360(s.hue - d.hue) < 180 ? 1 : -1;
  const hue = mod360(d.hue + rotation * direction);
  return { hex: hctToHex(hue, d.chroma, d.tone), hue, chroma: d.chroma };
}
