/** Браузерный бэкенд для `npm run dev` без Tauri: имитирует установку, запуск и выход игры. */
import { VALKYRIE_VERSION } from '../core/versions';
import type { Backend } from './api';
import type { GameExit, GameVersion, Settings, StatusReport, TaskEvent } from './types';

const sleep = (ms: number) => new Promise((resolve) => setTimeout(resolve, ms));

export function createMockBackend(): Backend {
  const stored = localStorage.getItem('valkyrie-mock-settings');
  const settings: Settings = {
    nickname: 'Player', manifestUrl: '', memoryMb: 4096, automaticMemory: true, selectedVersion: '1.21.10',
    selectedValkyrie: true, javaMode: 'automatic', customJava: '', theme: 'light', reducedMotion: false,
    uiSounds: true, changelogOpen: false, palette: 'sakura', ...(stored ? (JSON.parse(stored) as Partial<Settings>) : {}),
  };
  const installed = new Set<string>(JSON.parse(localStorage.getItem('valkyrie-mock-installed') ?? '[]') as string[]);
  const exitHandlers = new Set<(exit: GameExit) => void>();
  let selected: GameVersion = VALKYRIE_VERSION;
  const persist = () => {
    localStorage.setItem('valkyrie-mock-settings', JSON.stringify(settings));
    localStorage.setItem('valkyrie-mock-installed', JSON.stringify([...installed]));
  };
  const key = () => `${selected.valkyrie ? 'v' : 'm'}:${selected.id}`;
  const label = () => (selected.valkyrie ? 'Valkyrie 0.9.0-beta' : `Minecraft ${selected.id}`);
  const effective = () => (settings.automaticMemory ? 4096 : settings.memoryMb);
  const snapshot = () => ({ settings: { ...settings }, effectiveMemoryMb: effective() });
  const versions: GameVersion[] = [
    VALKYRIE_VERSION,
    ...['25w41a', '25w40a'].map((id) => ({ id, type: 'snapshot', metadataUrl: null, metadataSha1: '', valkyrie: false })),
    ...['1.21.10', '1.21.9', '1.21.8', '1.21.4', '1.21.1', '1.20.6', '1.20.4', '1.20.1', '1.19.4', '1.18.2', '1.16.5', '1.12.2', '1.8.9', '1.7.10']
      .map((id) => ({ id, type: 'release', metadataUrl: null, metadataSha1: '', valkyrie: false })),
  ];
  const simulate = async (onEvent: (event: TaskEvent) => void, quick = false) => {
    onEvent({ kind: 'progress', fraction: null });
    onEvent({ kind: 'status', message: `Подготавливаем Minecraft ${selected.id}` });
    await sleep(600);
    for (let i = 0; i <= 20; i++) {
      onEvent({ kind: 'status', message: `Скачиваем Minecraft и библиотеки · ${i * 4}/80` });
      onEvent({ kind: 'progress', fraction: 0.25 * (i / 20) });
      await sleep(quick ? 20 : 70);
    }
    for (let i = 0; i <= 40; i++) {
      onEvent({ kind: 'status', message: `Скачиваем игровые ресурсы · ${i * 98}/3920` });
      onEvent({ kind: 'progress', fraction: 0.25 + 0.65 * (i / 40) });
      await sleep(quick ? 15 : 60);
    }
    if (selected.valkyrie) {
      onEvent({ kind: 'progress', fraction: null });
      onEvent({ kind: 'status', message: 'Устанавливаем Forge 60.1.0' });
      await sleep(quick ? 600 : 2200);
    }
    onEvent({ kind: 'progress', fraction: 1 });
    installed.add(key());
    persist();
  };
  return {
    async bootstrap() {
      selected = settings.selectedValkyrie ? VALKYRIE_VERSION : versions.find((v) => !v.valkyrie && v.id === settings.selectedVersion) ?? VALKYRIE_VERSION;
      return {
        settings: { ...settings }, selected, launcherVersion: '2.0.0-dev', systemMemoryMb: 16384, effectiveMemoryMb: effective(),
        paths: { root: '%APPDATA%\\ValkyrieLauncher', game: '…\\game', instance: '…\\game', logs: '…\\logs' },
        accent: '#0078D4', bundledClientVersion: '0.9.0-beta', runningPid: null,
        defaultChangelog: [
          'Valkyrie Flow с единым Combat pipeline', 'Sakura Bloom, Threat Constellation и Memory Echo',
          'ClickGUI с анимированным Book Fold', 'Светлая и тёмная Sakura-темы', 'HUD Editor, Target HUD и Visual Profiles',
        ],
      };
    },
    async loadVersions() {
      await sleep(350);
      return { versions, fromCache: false };
    },
    async selectVersion(version) {
      selected = version;
      settings.selectedVersion = version.id;
      settings.selectedValkyrie = version.valkyrie;
      persist();
      return snapshot();
    },
    async checkStatus(): Promise<StatusReport> {
      await sleep(400);
      const ready = installed.has(key());
      return {
        state: ready ? 'READY' : 'INSTALL_REQUIRED', detail: selected.valkyrie ? label() : `Minecraft ${selected.id}`,
        manifest: null, manifestInstalled: false, changelog: null, warning: null, installedLabel: label(),
      };
    },
    async updateSettings(patch) {
      Object.assign(settings, patch);
      persist();
      return snapshot();
    },
    async setJava(mode, path) {
      settings.javaMode = mode;
      settings.customJava = mode === 'custom' ? (path ?? '') : '';
      persist();
      return snapshot();
    },
    async installAndLaunch(nickname, onEvent) {
      settings.nickname = nickname;
      persist();
      if (!installed.has(key())) await simulate(onEvent);
      await sleep(500);
      setTimeout(() => exitHandlers.forEach((handler) => handler({ code: 0, logPath: 'logs/minecraft-process.log', ready: true, installedLabel: label() })), 9000);
      return { pid: 4242, logPath: 'logs/minecraft-process.log', gameDirectory: '…\\game' };
    },
    async repair(onEvent) {
      installed.delete(key());
      await simulate(onEvent, true);
      return { ready: true, installedLabel: label() };
    },
    async openFolder(kind) {
      console.info(`[mock] open folder: ${kind}`);
    },
    async onGameExit(handler) {
      exitHandlers.add(handler);
      return () => exitHandlers.delete(handler);
    },
    async pickJava() {
      return 'C:\\Program Files\\Java\\jdk-21\\bin\\javaw.exe';
    },
    async systemAccent() {
      return '#0078D4';
    },
    async copyText(text) {
      await navigator.clipboard?.writeText(text);
    },
    log(level, message) {
      console[level](`[ui] ${message}`);
    },
    async checkLauncherUpdate() {
      await sleep(500);
      return null;
    },
    window: {
      startDragging: async () => undefined,
      minimize: async () => undefined,
      close: async () => window.close(),
      show: async () => undefined,
    },
  };
}
