import { useController, useUi } from '../app/hooks';
import { Button } from '../components/controls';
import { Icon } from '../components/Icon';
import { footerJava, footerPlatform } from '../core/versions';
import { t } from '../i18n/strings';

export function Footer() {
  const controller = useController();
  const selected = useUi((state) => state.selected);
  return (
    <footer className="footer">
      <div className="footer-info">
        <span className="chip">
          <Icon name={selected.valkyrie ? 'sakura' : 'cube'} size={16} />
          {footerPlatform(selected)}
        </span>
        <span className="chip" data-tone="java">
          <Icon name="java" size={16} />
          {footerJava(selected)}
        </span>
      </div>
      <div className="footer-actions">
        <Button variant="text" icon="folder" onClick={() => void controller.openFolder('game')}>
          {t.gameFolder}
        </Button>
        <Button variant="text" icon="logs" onClick={() => void controller.openFolder('logs')}>
          {t.logs}
        </Button>
      </div>
    </footer>
  );
}
