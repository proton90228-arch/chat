import { useController, useUi } from '../app/hooks';
import { IconButton } from '../components/controls';
import { Icon } from '../components/Icon';
import { t } from '../i18n/strings';

/** Своя строка заголовка: окно без рамки, перетаскивание за любую пустую область. */
export function TitleBar() {
  const controller = useController();
  const launcherVersion = useUi((state) => state.launcherVersion);
  const drawerOpen = useUi((state) => state.drawerOpen);
  const settingsOpen = useUi((state) => state.settingsOpen);
  const updateAvailable = useUi((state) => state.launcherUpdate.status === 'available');
  return (
    <header className="titlebar" data-tauri-drag-region>
      <div className="brand" data-tauri-drag-region>
        <span className="brand-mark">
          <Icon name="sakura" size={18} />
        </span>
        <span className="brand-name">Valkyrie</span>
        {launcherVersion && <span className="brand-version">· {launcherVersion}</span>}
      </div>
      <nav className="titlebar-actions" aria-label="Навигация">
        <IconButton icon="sparkles" label={t.updatesNav} selected={drawerOpen} onClick={() => controller.toggleDrawer()} />
        <IconButton
          icon="tune"
          label={t.settings}
          selected={settingsOpen}
          badge={updateAvailable}
          onClick={() => controller.toggleSettings()}
        />
        <span className="titlebar-divider" aria-hidden="true" />
        <IconButton icon="minimize" label={t.minimize} variant="window" onClick={() => controller.minimizeWindow()} />
        <IconButton icon="close" label={t.close} variant="danger" onClick={() => controller.closeWindow()} />
      </nav>
    </header>
  );
}
