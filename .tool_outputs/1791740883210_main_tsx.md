import '@fontsource-variable/inter';
import '@fontsource-variable/jost';
import '@fontsource-variable/jetbrains-mono';
import './styles/tokens.css';
import './styles/app.css';
import { StrictMode } from 'react';
import { createRoot } from 'react-dom/client';
import { App } from './App';
import { LauncherController } from './app/controller';
import { ControllerContext } from './app/hooks';
import { errorText } from './core/format';
import { getBackend } from './ipc/api';
import { SakuraSim } from './sakura/sim';
import { sounds } from './sound/sounds';

const backend = await getBackend();
const sim = new SakuraSim();
const controller = new LauncherController(backend, {
  gust: (strength) => sim.gust(strength),
  sound: (name) => sounds.play(name),
});

window.addEventListener('error', (event) => backend.log('error', `UI: ${event.message}`));
window.addEventListener('unhandledrejection', (event) => backend.log('error', `UI promise: ${errorText(event.reason)}`));

createRoot(document.getElementById('root')!).render(
  <StrictMode>
    <ControllerContext.Provider value={controller}>
      <App sim={sim} />
    </ControllerContext.Provider>
  </StrictMode>,
);

// Окно создаётся скрытым и показывается, когда тема уже применена: без белой вспышки.
let shown = false;
const show = () => {
  if (shown) return;
  shown = true;
  requestAnimationFrame(() =>
    requestAnimationFrame(() => {
      void backend.window.show().then(() => sounds.play('open'));
    }),
  );
};
const unsubscribe = controller.store.subscribe(() => {
  if (controller.state.phase !== 'booting') {
    unsubscribe();
    show();
  }
});
window.setTimeout(show, 2500);

void controller.start().then(() => controller.checkLauncherUpdate(true));
