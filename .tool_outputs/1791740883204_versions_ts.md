import { t } from '../i18n/strings';
import type { GameVersion } from '../ipc/types';

export const VALKYRIE_VERSION: GameVersion = {
  id: '1.21.10',
  type: 'release',
  metadataUrl: 'https://piston-meta.mojang.com/v1/packages/38e4f95edc94d09a3ded1e17a78b76f9694ca3b6/1.21.10.json',
  metadataSha1: '38e4f95edc94d09a3ded1e17a78b76f9694ca3b6',
  valkyrie: true,
};

export const versionKey = (v: GameVersion) => `${v.valkyrie ? 'valkyrie' : 'vanilla'}:${v.id}`;

export function displayName(v: GameVersion): string {
  return v.valkyrie ? '★ Valkyrie Client 1.21.10  ·  Recommended' : `Minecraft ${v.id}`;
}

/** Фильтр библиотеки версий: Valkyrie всегда видна, остальное — по типу и поиску. */
export function filterVersions(versions: readonly GameVersion[], query: string, snapshots: boolean): GameVersion[] {
  const needle = query.trim().toLowerCase();
  return versions.filter((v) => {
    if (v.valkyrie) return !needle || displayName(v).toLowerCase().includes(needle) || v.id.includes(needle);
    // Как в Java: вкладка Releases — только `release` (без old_beta/old_alpha), Snapshots — только `snapshot`.
    if (v.type !== (snapshots ? 'snapshot' : 'release')) return false;
    return !needle || v.id.toLowerCase().includes(needle) || displayName(v).toLowerCase().includes(needle);
  });
}

export const triggerTitle = (v: GameVersion) => (v.valkyrie ? t.valkyrieTitle : `Minecraft ${v.id}`);
export const triggerMeta = (v: GameVersion) =>
  v.valkyrie ? t.valkyrieMeta : v.type === 'snapshot' ? t.snapshotMeta : t.releaseMeta;
export const rowMeta = (v: GameVersion) =>
  v.valkyrie ? t.valkyrieRowMeta : v.type === 'snapshot' ? t.snapshotRowMeta : t.releaseRowMeta;
export const rowBadge = (v: GameVersion) => (v.valkyrie ? t.recommended : v.type.toUpperCase());
export const footerPlatform = (v: GameVersion) => (v.valkyrie ? t.valkyrieMeta : `Minecraft ${v.id} · Vanilla`);
export const footerJava = (v: GameVersion) => (v.valkyrie ? t.javaValkyrie : t.javaAuto);

/** Восстанавливает выбранную версию из каталога (нужна ссылка на метаданные). */
export function resolveSelected(versions: readonly GameVersion[], selected: GameVersion): GameVersion {
  return versions.find((v) => v.valkyrie === selected.valkyrie && v.id === selected.id) ?? selected;
}
