import { createContext, useContext, useEffect, useLayoutEffect, useRef, useState, useSyncExternalStore } from 'react';
import type { Store } from '../core/store';
import type { LauncherController, UiState } from './controller';

export const ControllerContext = createContext<LauncherController | null>(null);

export function useController(): LauncherController {
  const controller = useContext(ControllerContext);
  if (!controller) throw new Error('LauncherController не передан в ControllerContext');
  return controller;
}

/** Подписка на часть состояния. Селектор должен возвращать примитив или ссылку из состояния. */
export function useStore<T extends object, S>(store: Store<T>, selector: (state: T) => S): S {
  return useSyncExternalStore(store.subscribe, () => selector(store.get()), () => selector(store.get()));
}

export function useUi<S>(selector: (state: UiState) => S): S {
  return useStore(useController().store, selector);
}

export type PresenceState = 'enter' | 'open' | 'exit';

/**
 * Монтирование с анимацией выхода: `enter` (первый кадр) → `open` → `exit` → размонтирование.
 * CSS описывает конечное состояние для `open` и начальное для `enter`/`exit`.
 */
export function usePresence(open: boolean, exitMs: number): { mounted: boolean; state: PresenceState } {
  const [phase, setPhase] = useState<PresenceState | 'closed'>(open ? 'open' : 'closed');
  const [previous, setPrevious] = useState(open);
  // Смена `open` обрабатывается во время рендера (рекомендованный React способ «подстроить состояние под проп»).
  if (open !== previous) {
    setPrevious(open);
    setPhase(open ? 'enter' : 'exit');
  }
  useEffect(() => {
    if (phase === 'enter') {
      let inner = 0;
      const outer = requestAnimationFrame(() => {
        inner = requestAnimationFrame(() => setPhase('open'));
      });
      return () => {
        cancelAnimationFrame(outer);
        cancelAnimationFrame(inner);
      };
    }
    if (phase === 'exit') {
      const timer = window.setTimeout(() => setPhase('closed'), exitMs);
      return () => window.clearTimeout(timer);
    }
    return undefined;
  }, [phase, exitMs]);
  return { mounted: phase !== 'closed', state: phase === 'closed' ? 'exit' : phase };
}

/** Колбэк на каждый кадр; `dt` в секундах. */
export function useFrame(callback: (dt: number, now: number) => void, active = true): void {
  const saved = useRef(callback);
  useLayoutEffect(() => {
    saved.current = callback;
  });
  useEffect(() => {
    if (!active) return;
    let last = 0;
    let frame = requestAnimationFrame(function loop(now) {
      const dt = last === 0 ? 0 : (now - last) / 1000;
      last = now;
      saved.current(dt, now);
      frame = requestAnimationFrame(loop);
    });
    return () => cancelAnimationFrame(frame);
  }, [active]);
}
