import { describe, expect, it } from 'vitest';
import gameVersions from '../../contracts/game-versions.json';
import settings from '../../contracts/settings.json';
import statusReport from '../../contracts/status-report.json';
import taskEvents from '../../contracts/task-events.json';
import { errorText, isValidNickname, snapMemory } from '../src/core/format';
import { installBadge, LAUNCH_STATES, statusLine } from '../src/core/launchState';
import { ProgressModel } from '../src/core/progress';
import { filterVersions, VALKYRIE_VERSION } from '../src/core/versions';
import type { GameVersion, LaunchStateId, Settings, StatusReport, TaskEvent } from '../src/ipc/types';

describe('контракты IPC (те же JSON проверяет Rust: crates/valkyrie-core/tests/contracts.rs)', () => {
  it('совпадают с типами TypeScript', () => {
    const typedSettings: Settings = settings as Settings;
    const typedReport: StatusReport = statusReport as StatusReport;
    const typedEvents: TaskEvent[] = taskEvents as TaskEvent[];
    const typedVersions: GameVersion[] = gameVersions as GameVersion[];
    expect(Object.keys(typedSettings).sort()).toEqual(
      ['automaticMemory', 'changelogOpen', 'customJava', 'javaMode', 'manifestUrl', 'memoryMb', 'nickname', 'palette',
        'reducedMotion', 'selectedValkyrie', 'selectedVersion', 'theme', 'uiSounds'],
    );
    expect(LAUNCH_STATES[typedReport.state as LaunchStateId]).toBeDefined();
    expect(typedEvents.map((event) => event.kind)).toEqual(['status', 'stage', 'progress', 'progress']);
    expect(typedVersions[0]).toMatchObject({ id: '1.21.10', valkyrie: true });
  });
});

describe('LaunchState (как LaunchState.java)', () => {
  it('кнопка, статус и флаги', () => {
    expect(LAUNCH_STATES.READY).toMatchObject({ button: 'Запустить', actionable: true, busy: false, icon: 'play' });
    expect(LAUNCH_STATES.INSTALLING).toMatchObject({ actionable: false, busy: true });
    expect(LAUNCH_STATES.RUNNING).toMatchObject({ button: 'Запущено', actionable: false, busy: false, icon: 'stop' });
    expect(LAUNCH_STATES.ERROR).toMatchObject({ button: 'Повторить', actionable: true });
    expect(statusLine('READY', 'Valkyrie 0.9.0')).toBe('Готово · Valkyrie 0.9.0');
    expect(statusLine('CHECKING', '')).toBe('Проверка');
  });

  it('значок установки', () => {
    expect(installBadge('UPDATE_AVAILABLE', true, 'Valkyrie 0.9.0', '1.0.0').text).toBe('Доступно обновление · 1.0.0');
    expect(installBadge('INSTALL_REQUIRED', false, '', null).text).toBe('Не установлена');
    expect(installBadge('ERROR', true, '', null).text).toBe('Ошибка');
  });
});

describe('версии, никнейм, память, прогресс', () => {
  const versions: GameVersion[] = [
    VALKYRIE_VERSION,
    { id: '1.21.10', type: 'release', metadataUrl: 'u', metadataSha1: 's', valkyrie: false },
    { id: '25w41a', type: 'snapshot', metadataUrl: 'u', metadataSha1: 's', valkyrie: false },
    { id: '1.20.1', type: 'release', metadataUrl: 'u', metadataSha1: 's', valkyrie: false },
    { id: 'b1.7.3', type: 'old_beta', metadataUrl: 'u', metadataSha1: 's', valkyrie: false },
  ];

  it('фильтр Releases / Snapshots и поиск (Valkyrie видна всегда, как в Java)', () => {
    expect(filterVersions(versions, '', false).map((v) => v.id)).toEqual(['1.21.10', '1.21.10', '1.20.1']);
    expect(filterVersions(versions, '', true).map((v) => v.id)).toEqual(['1.21.10', '25w41a']);
    expect(filterVersions(versions, '1.20', false).map((v) => v.id)).toEqual(['1.20.1']);
  });

  it('никнейм 3–16 символов A-Z a-z 0-9 _', () => {
    expect(['Steve', 'sakura_17', 'abc', 'A234567890123456'].every(isValidNickname)).toBe(true);
    expect(['ab', 'A2345678901234567', 'с кириллицей', 'a-b'].some(isValidNickname)).toBe(false);
  });

  it('память: шаг 1024 в диапазоне 2048–8192', () => {
    expect([1000, 2600, 5000, 9000].map(snapMemory)).toEqual([2048, 3072, 5120, 8192]);
    expect(errorText(new Error('boom'))).toBe('boom');
  });

  it('прогресс монотонный, «ползёт» без процентов и не доходит до конца', () => {
    const progress = new ProgressModel();
    progress.set(0.5);
    progress.set(0.3);
    let value = 0;
    for (let i = 0; i < 120; i++) value = progress.tick(1 / 60);
    expect(value).toBeGreaterThan(0.49);
    progress.set(null);
    for (let i = 0; i < 60 * 600; i++) value = progress.tick(1 / 60);
    expect(value).toBeGreaterThan(0.5);
    expect(value).toBeLessThanOrEqual(0.995);
    progress.reset();
    expect(progress.tick(1 / 60, true)).toBe(0);
  });
});
