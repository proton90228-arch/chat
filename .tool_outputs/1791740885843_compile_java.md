import javax.tools.ToolProvider;
import java.nio.file.*;
import java.util.*;
public class Compile {
    public static void main(String[] a) throws Exception {
        List<String> args = new ArrayList<>(List.of("-d", "out", "-encoding", "UTF-8"));
        try (var s = Files.walk(Path.of("src"))) { s.filter(p -> p.toString().endsWith(".java")).forEach(p -> args.add(p.toString())); }
        int rc = ToolProvider.getSystemJavaCompiler().run(null, null, null, args.toArray(String[]::new));
        System.exit(rc);
    }
}
