import { useEffect, useRef } from 'react';
import { useController, useFrame, useUi } from '../app/hooks';
import { LAUNCH_STATES, playTone } from '../core/launchState';
import { Icon } from './Icon';

interface WaveColors {
  active: string;
  track: string;
  readAt: number;
}

/**
 * Волнистый линейный индикатор M3 Expressive. Активная часть — синусоида, бегущая вправо,
 * дорожка — прямая с точкой-ограничителем. В режиме reduced motion волна становится прямой.
 */
function drawWave(canvas: HTMLCanvasElement, value: number, now: number, flat: boolean, colors: WaveColors): void {
  const context = canvas.getContext('2d');
  if (!context) return;
  const ratio = window.devicePixelRatio || 1;
  const width = canvas.clientWidth;
  const height = canvas.clientHeight;
  if (canvas.width !== Math.round(width * ratio) || canvas.height !== Math.round(height * ratio)) {
    canvas.width = Math.round(width * ratio);
    canvas.height = Math.round(height * ratio);
  }
  if (now - colors.readAt > 400) {
    const style = getComputedStyle(canvas);
    colors.active = style.color;
    colors.track = style.getPropertyValue('--track').trim() || colors.track;
    colors.readAt = now;
  }
  context.setTransform(ratio, 0, 0, ratio, 0, 0);
  context.clearRect(0, 0, width, height);
  const stroke = 4;
  const mid = height / 2;
  const amplitude = flat ? 0 : 3;
  const wavelength = 34;
  const gap = 6;
  const start = stroke / 2;
  const finish = width - stroke / 2;
  const end = start + Math.max(0, Math.min(1, value)) * (finish - start);
  context.lineCap = 'round';
  context.lineWidth = stroke;

  const trackStart = end > start + 0.5 ? end + gap + stroke : start;
  if (trackStart < finish) {
    context.strokeStyle = colors.track;
    context.beginPath();
    context.moveTo(trackStart, mid);
    context.lineTo(finish, mid);
    context.stroke();
  }
  context.fillStyle = colors.active;
  context.beginPath();
  context.arc(finish, mid, stroke / 2, 0, Math.PI * 2);
  context.fill();

  if (end <= start + 0.5) return;
  const phase = (now / 1000) * Math.PI * 2 * 0.75;
  context.strokeStyle = colors.active;
  context.beginPath();
  for (let x = start; x <= end; x += 1) {
    const taper = Math.min(1, (x - start) / 14, (end - x) / 14 + 0.25);
    const y = mid + Math.sin(((x - start) / wavelength) * Math.PI * 2 - phase) * amplitude * taper;
    if (x === start) context.moveTo(x, y);
    else context.lineTo(x, y);
  }
  context.stroke();
}

/** Главная кнопка: состояние, смена иконки, пульс в работе, волнистый прогресс под ней. */
export function PlayButton() {
  const controller = useController();
  const launch = useUi((state) => state.launch);
  const reduced = useUi((state) => state.settings.reducedMotion);
  const info = LAUNCH_STATES[launch];
  const buttonRef = useRef<HTMLButtonElement>(null);
  const waveRef = useRef<HTMLCanvasElement>(null);
  const colors = useRef<WaveColors>({ active: '#A86279', track: '#FFD9E2', readAt: -1e9 });

  useEffect(() => {
    if (!info.busy) buttonRef.current?.style.setProperty('--progress', '0');
  }, [info.busy]);

  useFrame((dt, now) => {
    const value = controller.progress.tick(dt, reduced);
    buttonRef.current?.style.setProperty('--progress', value.toFixed(4));
    if (waveRef.current) drawWave(waveRef.current, value, now, reduced, colors.current);
  }, info.busy);

  return (
    <div className="play" data-tone={playTone(launch)} data-busy={info.busy} data-state={launch}>
      <button
        ref={buttonRef}
        type="button"
        className="play-button"
        disabled={!info.actionable}
        aria-busy={info.busy}
        onPointerEnter={() => controller.hoverGust()}
        onClick={() => void controller.launch()}
      >
        <span className="play-fill" aria-hidden="true" />
        <span className="play-icon" key={info.icon} data-icon={info.icon}>
          <Icon name={info.icon} size={22} />
        </span>
        <span className="play-label" key={launch}>
          {info.button}
        </span>
      </button>
      <canvas ref={waveRef} className="play-wave" data-visible={info.busy} aria-hidden="true" />
    </div>
  );
}
