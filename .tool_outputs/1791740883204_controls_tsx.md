import { useId, type ReactNode } from 'react';
import { Icon, type IconName } from './Icon';

/** Кнопки-иконки M3 (standard / tonal). */
export function IconButton(props: {
  icon: IconName;
  label: string;
  onClick: () => void;
  variant?: 'standard' | 'tonal' | 'window' | 'danger';
  selected?: boolean;
  disabled?: boolean;
  badge?: boolean;
}) {
  const { icon, label, onClick, variant = 'standard', selected = false, disabled = false, badge = false } = props;
  return (
    <button
      type="button"
      className="icon-button"
      data-variant={variant}
      aria-label={label}
      title={label}
      aria-pressed={variant === 'standard' ? selected : undefined}
      disabled={disabled}
      onClick={onClick}
    >
      <Icon name={icon} />
      {badge && <span className="icon-button-badge" aria-hidden="true" />}
    </button>
  );
}

export function Button(props: {
  children: ReactNode;
  onClick: () => void;
  variant?: 'filled' | 'tonal' | 'outlined' | 'text';
  icon?: IconName;
  disabled?: boolean;
  className?: string;
}) {
  const { children, onClick, variant = 'tonal', icon, disabled = false, className } = props;
  return (
    <button type="button" className={className ? `btn ${className}` : 'btn'} data-variant={variant} disabled={disabled} onClick={onClick}>
      {icon && <Icon name={icon} size={18} />}
      <span>{children}</span>
    </button>
  );
}

/** Переключатель M3: ручка растёт при включении, в ней появляется галочка. */
export function Switch(props: { checked: boolean; onChange: (checked: boolean) => void; label: string; disabled?: boolean }) {
  const { checked, onChange, label, disabled = false } = props;
  return (
    <button
      type="button"
      role="switch"
      className="switch"
      aria-checked={checked}
      aria-label={label}
      disabled={disabled}
      onClick={() => onChange(!checked)}
    >
      <span className="switch-handle">{checked && <Icon name="check" size={14} />}</span>
    </button>
  );
}

/** Ползунок с делениями (M3 Expressive: ручка-«палочка» с зазором). */
export function Slider(props: {
  value: number;
  min: number;
  max: number;
  step: number;
  onChange: (value: number) => void;
  disabled?: boolean;
  label: string;
  format: (value: number) => string;
}) {
  const { value, min, max, step, onChange, disabled = false, label, format } = props;
  const fraction = (value - min) / (max - min);
  const stops = Math.round((max - min) / step) + 1;
  return (
    <div className="slider" data-disabled={disabled} style={{ ['--fraction' as string]: fraction }}>
      <input
        type="range"
        min={min}
        max={max}
        step={step}
        value={value}
        disabled={disabled}
        aria-label={label}
        aria-valuetext={format(value)}
        onChange={(event) => onChange(Number(event.currentTarget.value))}
      />
      <div className="slider-track" aria-hidden="true">
        <span className="slider-active" />
        <span className="slider-inactive" />
        {Array.from({ length: stops }, (_, index) => (
          <span
            key={index}
            className="slider-stop"
            data-active={index / (stops - 1) < fraction}
            style={{ ['--at' as string]: index / (stops - 1) }}
          />
        ))}
        <span className="slider-handle" />
      </div>
    </div>
  );
}

/** Сегментированная кнопка M3 (одиночный выбор). */
export function Segmented<T extends string>(props: { value: T; options: readonly { id: T; label: string }[]; onChange: (id: T) => void; label: string }) {
  const { value, options, onChange, label } = props;
  return (
    <div className="segmented" role="radiogroup" aria-label={label}>
      {options.map((option) => (
        <button
          key={option.id}
          type="button"
          role="radio"
          aria-checked={option.id === value}
          className="segment"
          onClick={() => onChange(option.id)}
        >
          {option.id === value && <Icon name="check" size={16} />}
          <span>{option.label}</span>
        </button>
      ))}
    </div>
  );
}

/** Строка настройки: иконка в тональном круге, заголовок, подсказка, элемент управления. */
export function SettingRow(props: { icon: IconName; title: string; hint?: ReactNode; children?: ReactNode; htmlFor?: string }) {
  const { icon, title, hint, children } = props;
  const id = useId();
  return (
    <div className="setting-row" aria-labelledby={id}>
      <span className="setting-icon">
        <Icon name={icon} size={18} />
      </span>
      <span className="setting-text">
        <span className="setting-title" id={id}>
          {title}
        </span>
        {hint && <span className="setting-hint">{hint}</span>}
      </span>
      {children && <span className="setting-control">{children}</span>}
    </div>
  );
}
