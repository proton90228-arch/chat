import { useEffect, useMemo } from 'react';
import { useController, useUi } from './app/hooks';
import { SakuraField } from './components/SakuraField';
import { Toasts } from './components/Toasts';
import type { SakuraSim } from './sakura/sim';
import { ChangelogDrawer } from './screens/ChangelogDrawer';
import { FatalScreen } from './screens/FatalScreen';
import { Footer } from './screens/Footer';
import { LaunchCard } from './screens/LaunchCard';
import { SettingsPanel } from './screens/SettingsPanel';
import { TitleBar } from './screens/TitleBar';
import { VersionLibrary } from './screens/VersionLibrary';
import { sounds } from './sound/sounds';
import { applyTheme, petalColors } from './theme/theme';

/** Куда нельзя «хватать» окно: всё интерактивное и всплывающее (аналог `isInteractive` в Java). */
const NO_DRAG = 'button, input, a, label, [role="dialog"], [role="switch"], [role="option"], .toast, .scrim, .slider, [data-no-drag]';

export function App({ sim }: { sim: SakuraSim }) {
  const controller = useController();
  const phase = useUi((state) => state.phase);
  const dark = useUi((state) => state.settings.theme === 'dark');
  const palette = useUi((state) => state.settings.palette);
  const reducedMotion = useUi((state) => state.settings.reducedMotion);
  const uiSounds = useUi((state) => state.settings.uiSounds);
  const accent = useUi((state) => state.accent);
  const overlay = useUi((state) => state.libraryOpen || state.drawerOpen || state.settingsOpen);

  useEffect(() => {
    applyTheme({ palette, dark, accent, reducedMotion });
  }, [palette, dark, accent, reducedMotion]);

  useEffect(() => {
    sounds.enabled = uiSounds;
  }, [uiSounds]);

  const paint = useMemo(() => petalColors(palette, dark, accent), [palette, dark, accent]);

  useEffect(() => {
    const onKeyDown = (event: KeyboardEvent) => {
      if (event.key === 'Escape' && controller.closeTopmost()) event.preventDefault();
      // Звук печати: только печатные символы в полях ввода (LauncherSounds.type, 3 голоса).
      const target = event.target as HTMLElement | null;
      if (target?.tagName === 'INPUT' && event.key.length === 1 && !event.ctrlKey && !event.metaKey) sounds.play('type');
    };
    const onPointerDown = (event: PointerEvent) => {
      if (event.button !== 0) return;
      sounds.unlock();
      const target = event.target as Element | null;
      if (!target) return;
      const button = target.closest('button, [role="option"]');
      if (button && !button.classList.contains('sound-preview') && !(button as HTMLButtonElement).disabled) sounds.play('click');
      if (!target.closest(NO_DRAG)) controller.startDragging();
    };
    const onFocus = () => {
      void controller.refreshAccent();
    };
    window.addEventListener('keydown', onKeyDown);
    window.addEventListener('pointerdown', onPointerDown);
    window.addEventListener('focus', onFocus);
    return () => {
      window.removeEventListener('keydown', onKeyDown);
      window.removeEventListener('pointerdown', onPointerDown);
      window.removeEventListener('focus', onFocus);
    };
  }, [controller]);

  if (phase === 'fatal') return <FatalScreen />;

  return (
    <div className="app" data-phase={phase} data-overlay={overlay}>
      <div className="backdrop" aria-hidden="true">
        <span className="glow glow-a" />
        <span className="glow glow-b" />
      </div>
      <SakuraField sim={sim} paint={paint} reducedMotion={reducedMotion} />
      <div className="shell">
        <TitleBar />
        <div className="shell-body" inert={overlay || undefined}>
          <LaunchCard />
          <Footer />
        </div>
      </div>
      <ChangelogDrawer />
      <SettingsPanel />
      <VersionLibrary />
      <Toasts />
    </div>
  );
}
