import { useController, useUi } from '../app/hooks';
import { Icon } from '../components/Icon';
import { Logo } from '../components/Logo';
import { PlayButton } from '../components/PlayButton';
import { installBadge, statusDot, statusLine } from '../core/launchState';
import { triggerMeta, triggerTitle } from '../core/versions';
import { t } from '../i18n/strings';

function VersionTrigger() {
  const controller = useController();
  const selected = useUi((state) => state.selected);
  const launch = useUi((state) => state.launch);
  const installedLabel = useUi((state) => state.installedLabel);
  const manifestVersion = useUi((state) => state.manifestVersion);
  const disabled = !controller.canChangeVersion();
  const badge = installBadge(launch, selected.valkyrie, installedLabel, manifestVersion);
  return (
    <div className="version-group">
      <button
        type="button"
        className="version-trigger"
        data-valkyrie={selected.valkyrie}
        disabled={disabled}
        aria-haspopup="dialog"
        onClick={() => controller.setLibrary(true)}
      >
        <span className="version-icon">
          <Icon name={selected.valkyrie ? 'sakura' : 'cube'} size={22} />
        </span>
        <span className="version-text">
          <span className="version-title">{triggerTitle(selected)}</span>
          <span className="version-meta">{triggerMeta(selected)}</span>
        </span>
        {selected.valkyrie && <span className="badge">{t.recommended}</span>}
        <Icon name="chevronRight" size={20} className="version-chevron" />
      </button>
      {badge.text && (
        <span className="install-badge" data-kind={badge.kind}>
          <span className="install-dot" aria-hidden="true" />
          {badge.text}
        </span>
      )}
    </div>
  );
}

function NicknameField() {
  const controller = useController();
  const nickname = useUi((state) => state.nickname);
  const invalid = useUi((state) => state.nicknameInvalid);
  return (
    <label className="text-field" data-invalid={invalid}>
      <Icon name="person" size={20} className="text-field-icon" />
      <span className="text-field-body">
        <span className="text-field-label">{t.nickname}</span>
        <input
          value={nickname}
          maxLength={16}
          spellCheck={false}
          autoComplete="off"
          aria-invalid={invalid}
          aria-describedby="nickname-hint"
          onChange={(event) => controller.setNickname(event.currentTarget.value)}
          onKeyDown={(event) => {
            if (event.key === 'Enter') void controller.launch();
          }}
        />
      </span>
      <span className="text-field-counter" id="nickname-hint" title={t.nicknameHint}>
        {nickname.length}/16
      </span>
    </label>
  );
}

function StatusLine() {
  const launch = useUi((state) => state.launch);
  const detail = useUi((state) => state.detail);
  const validation = useUi((state) => state.validation);
  return (
    <div className="status" aria-live="polite">
      <p className="status-line" data-dot={statusDot(launch)}>
        <span className="status-dot" aria-hidden="true" />
        <span>{statusLine(launch, detail)}</span>
      </p>
      {validation && (
        <p className="validation" data-error={validation.error}>
          <Icon name={validation.error ? 'alert' : 'info'} size={16} />
          <span>{validation.text}</span>
        </p>
      )}
    </div>
  );
}

/** Герой-блок: логотип и карточка запуска (версия, никнейм, кнопка, статус). */
export function LaunchCard() {
  return (
    <main className="hero">
      <Logo className="hero-logo" />
      <section className="hero-card" aria-label="Запуск">
        <p className="hero-subtitle">{t.subtitle}</p>
        <VersionTrigger />
        <NicknameField />
        <PlayButton />
        <StatusLine />
      </section>
    </main>
  );
}
