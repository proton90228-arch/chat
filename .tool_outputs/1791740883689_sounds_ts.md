/** Звуки интерфейса (`LauncherSounds`): до 3 голосов на звук, не чаще одного раза в 48 мс. */
export type SoundName = 'click' | 'open' | 'close' | 'type';

const FILES: Record<SoundName, string> = {
  click: 'sounds/ui-click.wav',
  open: 'sounds/ui-open.wav',
  close: 'sounds/ui-close.wav',
  type: 'sounds/ui-type.wav',
};
const GAIN: Record<SoundName, number> = { click: 0.55, open: 0.5, close: 0.45, type: 0.32 };
const MAX_VOICES = 3;
const THROTTLE_MS = 48;

class Sounds {
  enabled = true;
  private context: AudioContext | null = null;
  private readonly buffers = new Map<SoundName, AudioBuffer>();
  private readonly voices = new Map<SoundName, AudioBufferSourceNode[]>();
  private readonly lastPlayed = new Map<SoundName, number>();
  private loading: Promise<void> | null = null;

  /** Создаёт AudioContext и декодирует звуки. Вызывается при первом жесте пользователя. */
  unlock(): void {
    if (typeof AudioContext === 'undefined') return;
    this.context ??= new AudioContext();
    if (this.context.state === 'suspended') void this.context.resume();
    this.loading ??= this.load(this.context);
  }

  private async load(context: AudioContext): Promise<void> {
    await Promise.all(
      (Object.keys(FILES) as SoundName[]).map(async (name) => {
        try {
          const response = await fetch(FILES[name]);
          this.buffers.set(name, await context.decodeAudioData(await response.arrayBuffer()));
        } catch {
          // Без звука — не ошибка.
        }
      }),
    );
  }

  play(name: SoundName): void {
    if (!this.enabled) return;
    this.unlock();
    const context = this.context;
    const buffer = this.buffers.get(name);
    if (!context || !buffer) return;
    const now = performance.now();
    if (now - (this.lastPlayed.get(name) ?? -Infinity) < THROTTLE_MS) return;
    this.lastPlayed.set(name, now);
    const voices = this.voices.get(name) ?? [];
    while (voices.length >= MAX_VOICES) {
      try {
        voices.shift()?.stop();
      } catch {
        // уже остановлен
      }
    }
    const source = context.createBufferSource();
    const gain = context.createGain();
    source.buffer = buffer;
    gain.gain.value = GAIN[name];
    source.connect(gain).connect(context.destination);
    source.onended = () => {
      const index = voices.indexOf(source);
      if (index >= 0) voices.splice(index, 1);
    };
    source.start();
    voices.push(source);
    this.voices.set(name, voices);
  }
}

export const sounds = new Sounds();
