import type {
  Bootstrap, Catalog, GameExit, GameVersion, JavaMode, LaunchInfo, RepairOutcome, SettingsPatch, SettingsSnapshot,
  StatusReport, TaskEvent,
} from './types';

export interface LauncherUpdate {
  version: string;
  notes: string;
  install(onProgress: (fraction: number | null) => void): Promise<void>;
}

export interface WindowControls {
  /** Перетаскивание окна за любую неинтерактивную область (как `NativeWindowDrag` в Java). */
  startDragging(): Promise<void>;
  minimize(): Promise<void>;
  close(): Promise<void>;
  show(): Promise<void>;
}

/** Всё, что интерфейс знает о бэкенде. Реализации: Tauri (`tauri.ts`) и браузерный mock (`mock.ts`). */
export interface Backend {
  bootstrap(): Promise<Bootstrap>;
  loadVersions(): Promise<Catalog>;
  selectVersion(version: GameVersion): Promise<SettingsSnapshot>;
  checkStatus(): Promise<StatusReport>;
  updateSettings(patch: SettingsPatch): Promise<SettingsSnapshot>;
  setJava(mode: JavaMode, path?: string | null): Promise<SettingsSnapshot>;
  installAndLaunch(nickname: string, onEvent: (event: TaskEvent) => void): Promise<LaunchInfo>;
  repair(onEvent: (event: TaskEvent) => void): Promise<RepairOutcome>;
  openFolder(kind: 'logs' | 'game'): Promise<void>;
  onGameExit(handler: (exit: GameExit) => void): Promise<() => void>;
  pickJava(): Promise<string | null>;
  /** Акцентный цвет Windows `#RRGGBB` для палитры «Windows»; `null`, если недоступен. */
  systemAccent(): Promise<string | null>;
  copyText(text: string): Promise<void>;
  log(level: 'info' | 'warn' | 'error', message: string): void;
  checkLauncherUpdate(): Promise<LauncherUpdate | null>;
  window: WindowControls;
}

export const isTauri = (): boolean => typeof window !== 'undefined' && '__TAURI_INTERNALS__' in window;

let backend: Promise<Backend> | null = null;

export function getBackend(): Promise<Backend> {
  backend ??= isTauri()
    ? import('./tauri').then((module) => module.createTauriBackend())
    : import('./mock').then((module) => module.createMockBackend());
  return backend;
}
