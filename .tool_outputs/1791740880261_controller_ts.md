/**
 * Оркестрация лаунчера — порт сценариев `ValkyrieLauncherApp` (Java) без привязки к React:
 * загрузка каталога, выбор версии, проверка обновлений, установка и запуск, починка, выход игры,
 * тосты, панели и настройки. Вся логика покрыта тестами `controller.test.ts`.
 */
import { t } from '../i18n/strings';
import { errorText, isValidNickname, snapMemory } from '../core/format';
import { LAUNCH_STATES } from '../core/launchState';
import { ProgressModel } from '../core/progress';
import { createStore, type Store } from '../core/store';
import { VALKYRIE_VERSION } from '../core/versions';
import type { Backend, LauncherUpdate } from '../ipc/api';
import type {
  Bootstrap, GameExit, GameVersion, LaunchStateId, Paths, Settings, SettingsPatch, SettingsSnapshot, TaskEvent,
} from '../ipc/types';
import type { SoundName } from '../sound/sounds';

export interface Toast {
  id: number;
  message: string;
  error: boolean;
  leaving: boolean;
}

export interface Validation {
  text: string;
  error: boolean;
}

export type UpdateStatus = 'idle' | 'checking' | 'latest' | 'available' | 'installing' | 'error';

export interface LauncherUpdateState {
  status: UpdateStatus;
  version: string | null;
  fraction: number | null;
}

export interface UiState {
  phase: 'booting' | 'ready' | 'fatal';
  fatal: string | null;
  launcherVersion: string;
  systemMemoryMb: number;
  bundledClientVersion: string;
  paths: Paths | null;
  settings: Settings;
  effectiveMemoryMb: number;
  accent: string | null;
  versions: GameVersion[];
  selected: GameVersion;
  launch: LaunchStateId;
  detail: string;
  installedLabel: string;
  manifestVersion: string | null;
  manifestInstalled: boolean;
  changes: string[];
  nickname: string;
  nicknameInvalid: boolean;
  validation: Validation | null;
  toasts: Toast[];
  settingsOpen: boolean;
  drawerOpen: boolean;
  libraryOpen: boolean;
  launcherUpdate: LauncherUpdateState;
}

/** Побочные эффекты интерфейса: порыв ветра у лепестков и звуки. */
export interface Effects {
  gust(strength: number): void;
  sound(name: SoundName): void;
}

export interface Clock {
  setTimeout(callback: () => void, ms: number): unknown;
}

export const TOAST_LIMIT = 3;
export const TOAST_MS = 3400;
export const TOAST_ERROR_MS = 5000;
export const TOAST_EXIT_MS = 180;

export const DEFAULT_SETTINGS: Settings = {
  nickname: 'Player',
  manifestUrl: '',
  memoryMb: 4096,
  automaticMemory: true,
  selectedVersion: '1.21.10',
  selectedValkyrie: true,
  javaMode: 'automatic',
  customJava: '',
  theme: 'light',
  reducedMotion: false,
  uiSounds: true,
  changelogOpen: false,
  palette: 'sakura',
};

export function initialState(): UiState {
  return {
    phase: 'booting',
    fatal: null,
    launcherVersion: '',
    systemMemoryMb: 0,
    bundledClientVersion: '',
    paths: null,
    settings: DEFAULT_SETTINGS,
    effectiveMemoryMb: 4096,
    accent: null,
    versions: [VALKYRIE_VERSION],
    selected: VALKYRIE_VERSION,
    launch: 'CHECKING',
    detail: '',
    installedLabel: '',
    manifestVersion: null,
    manifestInstalled: false,
    changes: [],
    nickname: 'Player',
    nicknameInvalid: false,
    validation: null,
    toasts: [],
    settingsOpen: false,
    drawerOpen: false,
    libraryOpen: false,
    launcherUpdate: { status: 'idle', version: null, fraction: null },
  };
}

export class LauncherController {
  readonly store: Store<UiState>;
  readonly progress = new ProgressModel();
  private toastSeq = 0;
  private statusSeq = 0;
  private pendingUpdate: LauncherUpdate | null = null;
  private unlistenExit: (() => void) | null = null;

  constructor(
    private readonly backend: Backend,
    private readonly effects: Effects,
    private readonly clock: Clock = globalThis,
  ) {
    this.store = createStore(initialState());
  }

  get state(): UiState {
    return this.store.get();
  }

  private set(patch: Partial<UiState>): void {
    this.store.set(patch);
  }

  // ---------------------------------------------------------------- запуск лаунчера

  async start(): Promise<void> {
    let boot: Bootstrap;
    try {
      boot = await this.backend.bootstrap();
    } catch (error) {
      this.backend.log('error', `bootstrap: ${errorText(error)}`);
      this.set({ phase: 'fatal', fatal: errorText(error) });
      return;
    }
    this.set({
      phase: 'ready',
      launcherVersion: boot.launcherVersion,
      systemMemoryMb: boot.systemMemoryMb,
      bundledClientVersion: boot.bundledClientVersion,
      paths: boot.paths,
      settings: boot.settings,
      effectiveMemoryMb: boot.effectiveMemoryMb,
      accent: boot.accent,
      selected: boot.selected,
      versions: boot.selected.valkyrie ? [VALKYRIE_VERSION] : [VALKYRIE_VERSION, boot.selected],
      nickname: boot.settings.nickname,
      nicknameInvalid: !isValidNickname(boot.settings.nickname),
      changes: boot.defaultChangelog,
      drawerOpen: boot.settings.changelogOpen,
    });
    try {
      this.unlistenExit = await this.backend.onGameExit((exit) => this.handleExit(exit));
    } catch (error) {
      this.backend.log('warn', `game-exited: ${errorText(error)}`);
    }
    if (boot.runningPid !== null) {
      // Интерфейс перезагрузился, а игра ещё идёт: только подтягиваем каталог.
      this.setLaunch('RUNNING', '');
      await this.loadVersions(false);
      return;
    }
    this.setLaunch('CHECKING', '');
    await this.loadVersions(true);
  }

  /** Лёгкий порыв ветра при наведении на «Запустить» (как в Java). */
  hoverGust(): void {
    if (LAUNCH_STATES[this.state.launch].actionable) this.effects.gust(0.35);
  }

  sound(name: SoundName): void {
    this.effects.sound(name);
  }

  minimizeWindow(): void {
    void this.backend.window.minimize();
  }

  /** Звук закрытия успевает прозвучать до закрытия окна (как `sounds.close()` в Java). */
  closeWindow(): void {
    this.effects.sound('close');
    this.clock.setTimeout(() => void this.backend.window.close(), 140);
  }

  startDragging(): void {
    void this.backend.window.startDragging();
  }

  async copyText(text: string): Promise<boolean> {
    try {
      await this.backend.copyText(text);
      return true;
    } catch {
      return false;
    }
  }

  dispose(): void {
    this.unlistenExit?.();
    this.unlistenExit = null;
  }

  /** `loadVersions`: каталог Mojang (или кэш); при ошибке — только Valkyrie и тост. */
  async loadVersions(select = true): Promise<void> {
    let versions: GameVersion[];
    let failed = false;
    try {
      versions = (await this.backend.loadVersions()).versions;
      if (!versions.some((version) => version.valkyrie)) versions = [VALKYRIE_VERSION, ...versions];
    } catch (error) {
      this.backend.log('warn', `Каталог версий: ${errorText(error)}`);
      versions = [VALKYRIE_VERSION];
      failed = true;
    }
    const { settings } = this.state;
    const restored = failed
      ? VALKYRIE_VERSION
      : (versions.find((v) => v.valkyrie === settings.selectedValkyrie && (v.valkyrie || v.id === settings.selectedVersion)) ??
        VALKYRIE_VERSION);
    this.set({ versions });
    if (failed) this.toast(t.catalogUnavailable, true);
    if (select) await this.selectVersion(restored, false);
  }

  // ---------------------------------------------------------------- версия и статус

  /** Выбор версии: сохраняем в конфиг и сразу проверяем установку и обновления. */
  async selectVersion(version: GameVersion, userAction = true): Promise<void> {
    if (userAction && !this.canChangeVersion()) return;
    this.set({ selected: version, libraryOpen: false });
    try {
      this.applySnapshot(await this.backend.selectVersion(version));
    } catch (error) {
      this.toast(errorText(error), true);
    }
    await this.checkStatus();
  }

  canChangeVersion(): boolean {
    const { launch } = this.state;
    return !LAUNCH_STATES[launch].busy && launch !== 'RUNNING';
  }

  /** `checkUpdates`. На время проверки кнопка заблокирована (в Java здесь была гонка). */
  async checkStatus(): Promise<void> {
    const request = ++this.statusSeq;
    this.setLaunch('CHECKING', '');
    try {
      const report = await this.backend.checkStatus();
      if (request !== this.statusSeq) return;
      this.set({
        manifestVersion: report.manifest?.version ?? null,
        manifestInstalled: report.manifestInstalled,
        installedLabel: report.installedLabel,
        ...(report.changelog && report.changelog.length > 0 ? { changes: report.changelog } : {}),
      });
      this.setLaunch(report.state, report.detail);
      if (report.warning) {
        this.showValidation(report.warning, false);
        this.toast(t.updateServerDown, false);
      }
    } catch (error) {
      if (request !== this.statusSeq) return;
      this.setLaunch('ERROR', errorText(error));
    }
  }

  /** `setLaunchState`: вне «занятых» состояний прогресс сбрасывается. */
  private setLaunch(launch: LaunchStateId, detail: string): void {
    if (!LAUNCH_STATES[launch].busy) this.progress.reset();
    this.set({ launch, detail });
  }

  // ---------------------------------------------------------------- установка и запуск

  private taskEvents = (event: TaskEvent): void => {
    switch (event.kind) {
      case 'status':
        this.setLaunch('INSTALLING', event.message);
        break;
      case 'stage':
        // Новая фаза (например, загрузка Java): полоса начинает с нуля, а не стоит на 100 %.
        this.progress.reset();
        this.setLaunch('INSTALLING', event.message);
        break;
      case 'progress':
        this.progress.set(event.fraction);
        break;
    }
  };

  /** `installAndLaunch`: никнейм → установка/обновление → запуск → ожидание выхода игры. */
  async launch(): Promise<void> {
    const state = this.state;
    if (!LAUNCH_STATES[state.launch].actionable) return;
    if (!this.validateNickname(true)) return;
    const nickname = state.nickname.trim();
    const detail = state.detail;
    const updating = state.selected.valkyrie && state.manifestVersion !== null && !state.manifestInstalled;
    this.effects.gust(1.0);
    this.set({ nickname, settings: { ...state.settings, nickname } });
    this.progress.reset();
    this.setLaunch(updating ? 'UPDATING' : 'STARTING', detail);
    try {
      await this.backend.installAndLaunch(nickname, this.taskEvents);
      this.set({ manifestInstalled: true });
      this.setLaunch('RUNNING', detail);
      this.toast(t.launched, false);
    } catch (error) {
      const message = errorText(error);
      this.backend.log('error', `Ошибка установки или запуска Minecraft: ${message}`);
      this.setLaunch('ERROR', t.launchError);
      this.showValidation(message, true);
      this.toast(message, true);
    }
  }

  private handleExit(exit: GameExit): void {
    this.set({ installedLabel: exit.installedLabel });
    if (exit.code === 0) {
      this.setLaunch('READY', exit.installedLabel);
      return;
    }
    this.setLaunch('ERROR', t.exitedWithCode(exit.code));
    this.showValidation(t.exitedLog(exit.logPath), true);
    this.toast(t.exitedWithError(exit.code), true);
  }

  /** «Verify and repair»: переустановка с проверкой хэшей. */
  async repair(): Promise<void> {
    const { launch, selected } = this.state;
    if (LAUNCH_STATES[launch].busy || launch === 'RUNNING') return;
    this.set({ settingsOpen: false });
    this.progress.reset();
    this.setLaunch('INSTALLING', t.checkingFiles);
    try {
      const outcome = await this.backend.repair(this.taskEvents);
      this.set({ installedLabel: outcome.installedLabel });
      this.setLaunch(outcome.ready ? 'READY' : 'INSTALL_REQUIRED', outcome.installedLabel);
      this.toast(selected.valkyrie ? t.readyValkyrie : t.readyVanilla, false);
    } catch (error) {
      const message = errorText(error);
      this.backend.log('error', `Ошибка проверки установки: ${message}`);
      this.setLaunch('ERROR', t.installError);
      this.showValidation(message, true);
      this.toast(message, true);
    }
  }

  // ---------------------------------------------------------------- никнейм и сообщения

  setNickname(value: string): void {
    const valid = isValidNickname(value);
    this.set({ nickname: value, nicknameInvalid: !valid, ...(valid ? { validation: null } : {}) });
  }

  /** `validateNickname`: подсветка поля всегда, текст ошибки — только при попытке запуска. */
  validateNickname(showError: boolean): boolean {
    const valid = isValidNickname(this.state.nickname);
    this.set({ nicknameInvalid: !valid });
    if (showError && !valid) this.showValidation(t.nicknameHint, true);
    else if (valid) this.set({ validation: null });
    return valid;
  }

  showValidation(text: string, error: boolean): void {
    this.set({ validation: { text: text.trim() ? text : t.unknownError, error } });
  }

  /** Тосты: не больше трёх, новые сверху, ошибки живут дольше. */
  toast(message: string, error: boolean): void {
    const id = ++this.toastSeq;
    const toasts = [{ id, message: message.trim() ? message : t.unknownError, error, leaving: false }, ...this.state.toasts];
    this.set({ toasts: toasts.slice(0, TOAST_LIMIT) });
    this.clock.setTimeout(() => this.dismissToast(id), error ? TOAST_ERROR_MS : TOAST_MS);
  }

  dismissToast(id: number): void {
    const toast = this.state.toasts.find((item) => item.id === id);
    if (!toast || toast.leaving) return;
    this.set({ toasts: this.state.toasts.map((item) => (item.id === id ? { ...item, leaving: true } : item)) });
    this.clock.setTimeout(() => this.set({ toasts: this.state.toasts.filter((item) => item.id !== id) }), TOAST_EXIT_MS);
  }

  // ---------------------------------------------------------------- панели

  toggleSettings(): void {
    const open = !this.state.settingsOpen;
    if (open && this.state.drawerOpen) this.setDrawer(false);
    this.set({ settingsOpen: open });
  }

  toggleDrawer(): void {
    this.setDrawer(!this.state.drawerOpen);
  }

  setDrawer(open: boolean): void {
    if (open && this.state.settingsOpen) this.set({ settingsOpen: false });
    if (open === this.state.drawerOpen) return;
    this.set({ drawerOpen: open });
    void this.patchSettings({ changelogOpen: open });
  }

  setLibrary(open: boolean): void {
    if (open === this.state.libraryOpen) return;
    if (open) {
      if (!this.canChangeVersion()) return;
      if (this.state.drawerOpen) this.setDrawer(false);
      this.set({ settingsOpen: false });
    }
    this.set({ libraryOpen: open });
  }

  /** ESC: библиотека версий → changelog → настройки. */
  closeTopmost(): boolean {
    const { libraryOpen, drawerOpen, settingsOpen } = this.state;
    if (libraryOpen) this.setLibrary(false);
    else if (drawerOpen) this.setDrawer(false);
    else if (settingsOpen) this.set({ settingsOpen: false });
    else return false;
    return true;
  }

  // ---------------------------------------------------------------- настройки

  private applySnapshot(snapshot: SettingsSnapshot): void {
    this.set({ settings: snapshot.settings, effectiveMemoryMb: snapshot.effectiveMemoryMb });
  }

  async patchSettings(patch: SettingsPatch): Promise<void> {
    this.set({ settings: { ...this.state.settings, ...patch } });
    try {
      this.applySnapshot(await this.backend.updateSettings(patch));
    } catch (error) {
      this.backend.log('error', `Не удалось сохранить настройки: ${errorText(error)}`);
      this.showValidation(t.saveFailed, true);
    }
  }

  setDark(dark: boolean): Promise<void> {
    return this.patchSettings({ theme: dark ? 'dark' : 'light' });
  }

  setPalette(palette: string): Promise<void> {
    return this.patchSettings({ palette });
  }

  setReducedMotion(reducedMotion: boolean): Promise<void> {
    return this.patchSettings({ reducedMotion });
  }

  setUiSounds(uiSounds: boolean): Promise<void> {
    const done = this.patchSettings({ uiSounds });
    if (uiSounds) this.effects.sound('click');
    return done;
  }

  setAutomaticMemory(automaticMemory: boolean): Promise<void> {
    return this.patchSettings({ automaticMemory });
  }

  setMemory(memoryMb: number): Promise<void> {
    const snapped = snapMemory(memoryMb);
    if (snapped === this.state.settings.memoryMb) return Promise.resolve();
    return this.patchSettings({ memoryMb: snapped });
  }

  /** «Change» у Java: своя Java ↔ автоматическая. */
  async chooseJava(): Promise<void> {
    try {
      if (this.state.settings.javaMode === 'custom') {
        this.applySnapshot(await this.backend.setJava('automatic'));
        return;
      }
      const path = await this.backend.pickJava();
      if (!path) return;
      this.applySnapshot(await this.backend.setJava('custom', path));
    } catch (error) {
      this.showValidation(errorText(error), true);
      this.toast(errorText(error), true);
    }
  }

  async openFolder(kind: 'logs' | 'game'): Promise<void> {
    try {
      await this.backend.openFolder(kind);
    } catch (error) {
      this.backend.log('warn', `open_folder(${kind}): ${errorText(error)}`);
      this.toast(t.folderFailed, true);
    }
  }

  /** Акцент Windows мог измениться, пока лаунчер был в фоне. */
  async refreshAccent(): Promise<void> {
    try {
      const accent = await this.backend.systemAccent();
      if (accent !== this.state.accent) this.set({ accent });
    } catch {
      // акцент не обязателен
    }
  }

  // ---------------------------------------------------------------- обновление самого лаунчера

  async checkLauncherUpdate(silent = false): Promise<void> {
    const { status } = this.state.launcherUpdate;
    if (status === 'checking' || status === 'installing') return;
    this.set({ launcherUpdate: { status: 'checking', version: null, fraction: null } });
    try {
      const update = await this.backend.checkLauncherUpdate();
      this.pendingUpdate = update;
      if (!update) {
        this.set({ launcherUpdate: { status: 'latest', version: null, fraction: null } });
        if (!silent) this.toast(t.upToDate, false);
        return;
      }
      this.set({ launcherUpdate: { status: 'available', version: update.version, fraction: null } });
      this.toast(t.updateFound(update.version), false);
    } catch (error) {
      this.backend.log('warn', `Проверка обновлений лаунчера: ${errorText(error)}`);
      this.set({ launcherUpdate: { status: silent ? 'idle' : 'error', version: null, fraction: null } });
      if (!silent) this.toast(t.updateFailed, true);
    }
  }

  async installLauncherUpdate(): Promise<void> {
    const update = this.pendingUpdate;
    if (!update || LAUNCH_STATES[this.state.launch].busy) return;
    this.set({ launcherUpdate: { status: 'installing', version: update.version, fraction: null } });
    try {
      // Установщик NSIS в режиме passive сам закроет и перезапустит лаунчер.
      await update.install((fraction) =>
        this.set({ launcherUpdate: { status: 'installing', version: update.version, fraction } }),
      );
    } catch (error) {
      this.backend.log('error', `Обновление лаунчера: ${errorText(error)}`);
      this.set({ launcherUpdate: { status: 'available', version: update.version, fraction: null } });
      this.toast(t.updateFailed, true);
    }
  }
}
