#!/usr/bin/env sh
# Пересобирает эталон ui/tests/fixtures/sakura-golden.json из ОРИГИНАЛЬНОГО SakuraBackground.java.
# Нужен только JDK 17+ (javac не обязателен: компиляция идёт через javax.tools из Compile.java).
#   sh tools/golden/run.sh path/to/valkyrieclient/launcher/src/main/java/com/valkyrie/launcher/SakuraBackground.java
set -eu
SOURCE="${1:?Укажите путь к SakuraBackground.java из Java-лаунчера}"
HERE="$(cd "$(dirname "$0")" && pwd)"
ROOT="$(cd "$HERE/../.." && pwd)"
BUILD="$HERE/build"
rm -rf "$BUILD"
mkdir -p "$BUILD/src/com/valkyrie/launcher"
cp -R "$HERE/javafx" "$HERE/stub" "$BUILD/src/"
cp "$HERE/harness/GoldenHarness.java" "$BUILD/src/com/valkyrie/launcher/"
cp "$SOURCE" "$BUILD/src/com/valkyrie/launcher/SakuraBackground.java"
cd "$BUILD"
java "$HERE/Compile.java"
java -cp out com.valkyrie.launcher.GoldenHarness > "$ROOT/ui/tests/fixtures/sakura-golden.json"
echo "OK: ui/tests/fixtures/sakura-golden.json обновлён"
