import { buildScheme, PRESETS, type PresetId } from './scheme';

export type PaletteId = PresetId | 'windows';

export const isPreset = (id: string): id is PresetId => Object.hasOwn(PRESETS, id);

const VK_ROLE = /^((on-)?(install|update|success)(-container)?|petal|petal-soft|brand|on-brand)$/;

/** CSS-переменная роли: стандартные роли M3 — `--md-sys-color-*`, роли Valkyrie — `--vk-*`. */
export const cssVariable = (role: string) => (VK_ROLE.test(role) ? `--vk-${role}` : `--md-sys-color-${role}`);

export interface ThemeOptions {
  palette: string;
  dark: boolean;
  accent: string | null;
  reducedMotion: boolean;
}

let dynamicVariables: string[] = [];

/**
 * Пресеты берутся из статических токенов (`data-palette`), палитра «Windows» строится на лету
 * из системного акцента тем же алгоритмом Sakura You.
 */
export function applyTheme(options: ThemeOptions, root: HTMLElement = document.documentElement): void {
  root.dataset.theme = options.dark ? 'dark' : 'light';
  root.dataset.reducedMotion = String(options.reducedMotion);
  for (const name of dynamicVariables) root.style.removeProperty(name);
  dynamicVariables = [];
  if (options.palette === 'windows' && options.accent) {
    root.dataset.palette = 'windows';
    const scheme = buildScheme(options.accent)[options.dark ? 'dark' : 'light'];
    for (const [role, hex] of Object.entries(scheme)) {
      const name = cssVariable(role);
      root.style.setProperty(name, hex);
      dynamicVariables.push(name);
    }
  } else {
    root.dataset.palette = isPreset(options.palette) ? options.palette : 'sakura';
  }
  root.style.colorScheme = options.dark ? 'dark' : 'light';
}

export interface PetalColors {
  pale: string;
  deep: string;
  alpha: number;
}

/** Цвета лепестков. Для «Сакуры» — точные цвета Java-версии. */
export function petalColors(palette: string, dark: boolean, accent: string | null): PetalColors {
  const seed = palette === 'windows' ? accent : isPreset(palette) && palette !== 'sakura' ? PRESETS[palette].seed : null;
  if (!seed) {
    return dark ? { pale: '#F7DDE8', deep: '#E9A9C2', alpha: 0.62 } : { pale: '#F2CEDE', deep: '#E8B4C8', alpha: 0.55 };
  }
  const scheme = buildScheme(seed)[dark ? 'dark' : 'light'];
  return dark
    ? { pale: scheme['primary-fixed']!, deep: scheme.petal!, alpha: 0.62 }
    : { pale: scheme['petal-soft']!, deep: scheme.petal!, alpha: 0.55 };
}
